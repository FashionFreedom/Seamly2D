  set(QT_QTXML Qt6::Xml)
  set(QT_QTSVG Qt6::Svg)
  set(QT_QTPRINTSUPPORT Qt6::PrintSupport)
  set(QT_QTNETWORK Qt6::Network)
  set(QT_QTMULTIMEDIA Qt6::Multimedia)

  set(QT_QTGUI Qt5::Gui)
  set(QT_QTWIDGETS Qt5::Widgets)
  set(QT_QTXML Qt5::Xml)
  set(QT_QTSVG Qt5::Svg)
  set(QT_QTPRINTSUPPORT Qt5::PrintSupport)
  set(QT_QTNETWORK Qt5::Network)
  set(QT_QTMULTIMEDIA Qt5::Multimedia)
  set(QT_QTTEST Qt5::Test)
endif()

function(qt_targets result)
  set(qt_targets_list "")
  foreach(mod IN LISTS ARGN)
    if(mod STREQUAL "Core")
      list(APPEND qt_targets_list ${QT_QTCORE})
    elseif(mod STREQUAL "Gui")
      list(APPEND qt_targets_list ${QT_QTGUI})
    elseif(mod STREQUAL "Widgets")
      list(APPEND qt_targets_list ${QT_QTWIDGETS})
    elseif(mod STREQUAL "Xml")
      list(APPEND qt_targets_list ${QT_QTXML})
    elseif(mod STREQUAL "Svg")
      list(APPEND qt_targets_list ${QT_QTSVG})
    elseif(mod STREQUAL "PrintSupport")
      list(APPEND qt_targets_list ${QT_QTPRINTSUPPORT})
    elseif(mod STREQUAL "Network")
      list(APPEND qt_targets_list ${QT_QTNETWORK})
    elseif(mod STREQUAL "Multimedia")
      list(APPEND qt_targets_list ${QT_QTMULTIMEDIA})
    elseif(mod STREQUAL "Test")
      list(APPEND qt_targets_list ${QT_QTTEST})
    endif()
  endforeach()
  set(${result} "${qt_targets_list}" PARENT_SCOPE)
endfunction()

find_package(Git QUIET)
set(BUILD_REVISION "unknown")
if(GIT_FOUND)
  execute_process(
    COMMAND "${GIT_EXECUTABLE}" rev-parse --short=12 HEAD
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    OUTPUT_VARIABLE GIT_COMMIT
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
  )
  if(GIT_COMMIT)
    set(BUILD_REVISION "Git:${GIT_COMMIT}")
  endif()
endif()

add_compile_definitions(QT_MESSAGELOGCONTEXT)
add_compile_definitions(BUILD_REVISION="${BUILD_REVISION}")
if(NOT CMAKE_BUILD_TYPE STREQUAL "Debug")
  add_compile_definitions(QT_NO_DEBUG_OUTPUT)
endif()
