93/**
94 * @brief CutSpline cut spline.
95 * @param length length first spline
96 * @param spl1p2 second point of first spline
97 * @param spl1p3 third point of first spline
98 * @param spl2p2 second point of second spline
99 * @param spl2p3 third point of second spline
100 * @return point of cutting. This point is forth point of first spline and first point of second spline.
101 */
102QPointF VAbstractCubicBezier::CutSpline(qreal length, QPointF &spl1p2, QPointF &spl1p3, QPointF &spl2p2,
103 QPointF &spl2p3) const
104{
105 //Always need return two splines, so we must correct wrong length.
106 const qreal minLength = ToPixel(1, Unit::Mm);
107 const qreal fullLength = GetLength();
108
109 if (fullLength <= minLength)
110 {
111 spl1p2 = spl1p3 = spl2p2 = spl2p3 = QPointF();
112 return QPointF();
113 }
114
115 const qreal maxLength = fullLength - minLength;
116
117 if (length < minLength)
118 {
119 length = minLength;
120 }
121 else if (length > maxLength)
122 {
123 length = maxLength;
124 }
125
126 const qreal parT = GetParmT(length);
127
128 QLineF seg1_2 ( static_cast<QPointF>(GetP1 ()), GetControlPoint1 () );
129 seg1_2.setLength(seg1_2.length () * parT);
130 const QPointF p12 = seg1_2.p2();
131
132 QLineF seg2_3 ( GetControlPoint1(), GetControlPoint2 () );
133 seg2_3.setLength(seg2_3.length () * parT);
134 const QPointF p23 = seg2_3.p2();
135
136 QLineF seg12_23 ( p12, p23 );
137 seg12_23.setLength(seg12_23.length () * parT);
138 const QPointF p123 = seg12_23.p2();
139
140 QLineF seg3_4 ( GetControlPoint2 (), static_cast<QPointF>(GetP4 ()) );
141 seg3_4.setLength(seg3_4.length () * parT);
142 const QPointF p34 = seg3_4.p2();
143
144 QLineF seg23_34 ( p23, p34 );
145 seg23_34.setLength(seg23_34.length () * parT);
146 const QPointF p234 = seg23_34.p2();
147
148 QLineF seg123_234 ( p123, p234 );
149 seg123_234.setLength(seg123_234.length () * parT);
150 const QPointF p1234 = seg123_234.p2();
151
152 spl1p2 = p12;
153 spl1p3 = p123;
154 spl2p2 = p234;
155 spl2p3 = p34;
156 return p1234;
157}

170//---------------------------------------------------------------------------------------------------------------------
171qreal VAbstractCubicBezier::GetParmT(qreal length) const
172{
173 if (length < 0)
174 {
175 return 0;
176 }
177 else if (length > GetLength())
178 {
179 length = GetLength();
180 }
181
182 const qreal eps = 0.001 * length;
183 qreal parT = 0.5;
184 qreal step = parT;
185 qreal splLength = LengthT(parT);
186
187 while (qAbs(splLength - length) > eps)
188 {
189 step /= 2.0;
190 splLength > length ? parT -= step : parT += step;
191 splLength = LengthT(parT);
192 }
193 return parT;
194}

209/**
210 * @brief CalcSqDistance calculate squared distance.
211 * @param x1 х coordinate first point.
212 * @param y1 у coordinate first point.
213 * @param x2 х coordinate second point.
214 * @param y2 у coordinate second point.
215 * @return squared length.
216 */
217qreal VAbstractCubicBezier::CalcSqDistance(qreal x1, qreal y1, qreal x2, qreal y2)
218{
219 const qreal dx = x2 - x1;
220 const qreal dy = y2 - y1;
221 return dx * dx + dy * dy;
222}

225/**
226 * @brief PointBezier_r find spline point using four point of spline.
227 * @param x1 х coordinate first point.
228 * @param y1 у coordinate first point.
229 * @param x2 х coordinate first control point.
230 * @param y2 у coordinate first control point.
231 * @param x3 х coordinate second control point.
232 * @param y3 у coordinate second control point.
233 * @param x4 х coordinate last point.
234 * @param y4 у coordinate last point.
235 * @param level level of recursion. In the begin 0.
236 * @param px list х coordinat spline points.
237 * @param py list у coordinat spline points.
238 */
239void VAbstractCubicBezier::PointBezier_r(qreal x1, qreal y1, qreal x2, qreal y2, qreal x3, qreal y3, qreal x4, qreal y4,
240 qint16 level, QVector<qreal> &px, QVector<qreal> &py)
241{
242 if (px.size() >= 2)
243 {
244 for (int i=1; i < px.size(); ++i)
245 {
246 if (QPointF(px.at(i-1), py.at(i-1)) == QPointF(px.at(i), py.at(i)))
247 {
248 qDebug("All neighbors points in path must be unique.");
249 }
250 }
251 }
252
253 const double curve_collinearity_epsilon = 1e-30;
254 const double curve_angle_tolerance_epsilon = 0.01;
255 const double m_angle_tolerance = 0.0;
256 enum curve_recursion_limit_e { curve_recursion_limit = 32 };
257 const double m_cusp_limit = 0.0;
258 double m_approximation_scale = 1.0;
259 double m_distance_tolerance_square;
260
261 m_distance_tolerance_square = 0.5 / m_approximation_scale;
262 m_distance_tolerance_square *= m_distance_tolerance_square;
263
264 if (level > curve_recursion_limit)
265 {
266 return;
267 }
268
269 // Calculate all the mid-points of the line segments
270 //----------------------
271 const double x12 = (x1 + x2) / 2;
272 const double y12 = (y1 + y2) / 2;
273 const double x23 = (x2 + x3) / 2;
274 const double y23 = (y2 + y3) / 2;
275 const double x34 = (x3 + x4) / 2;
276 const double y34 = (y3 + y4) / 2;
277 const double x123 = (x12 + x23) / 2;
278 const double y123 = (y12 + y23) / 2;
279 const double x234 = (x23 + x34) / 2;
280 const double y234 = (y23 + y34) / 2;
281 const double x1234 = (x123 + x234) / 2;
282 const double y1234 = (y123 + y234) / 2;
283
284
285 // Try to approximate the full cubic curve by a single straight line
286 //------------------
287 const double dx = x4-x1;
288 const double dy = y4-y1;
289
290 double d2 = fabs((x2 - x4) * dy - (y2 - y4) * dx);
291 double d3 = fabs((x3 - x4) * dy - (y3 - y4) * dx);
292
293 switch ((static_cast<int>(d2 > curve_collinearity_epsilon) << 1) +
294 static_cast<int>(d3 > curve_collinearity_epsilon))
295 {
296 case 0:
297 {
298 // All collinear OR p1==p4
299 //----------------------
300 double k = dx*dx + dy*dy;
301 if (k < 0.000000001)
302 {
303 d2 = CalcSqDistance(x1, y1, x2, y2);
304 d3 = CalcSqDistance(x4, y4, x3, y3);
305 }
306 else
307 {
308 k = 1 / k;
309 {
310 const double da1 = x2 - x1;
311 const double da2 = y2 - y1;
312 d2 = k * (da1*dx + da2*dy);
313 }
314 {
315 const double da1 = x3 - x1;
316 const double da2 = y3 - y1;
317 d3 = k * (da1*dx + da2*dy);
318 }
319 if (d2 > 0 && d2 < 1 && d3 > 0 && d3 < 1)
320 {
321 // Simple collinear case, 1---2---3---4
322 // We can leave just two endpoints
323 return;
324 }
325 if (d2 <= 0)
326 {
327 d2 = CalcSqDistance(x2, y2, x1, y1);
328 }
329 else if (d2 >= 1)
330 {
331 d2 = CalcSqDistance(x2, y2, x4, y4);
332 }
333 else
334 {
335 d2 = CalcSqDistance(x2, y2, x1 + d2*dx, y1 + d2*dy);
336 }
337
338 if (d3 <= 0)
339 {
340 d3 = CalcSqDistance(x3, y3, x1, y1);
341 }
342 else if (d3 >= 1)
343 {
344 d3 = CalcSqDistance(x3, y3, x4, y4);
345 }
346 else
347 {
348 d3 = CalcSqDistance(x3, y3, x1 + d3*dx, y1 + d3*dy);
349 }
350 }
351 if (d2 > d3)
352 {
353 if (d2 < m_distance_tolerance_square)
354 {
355 px.append(x2);
356 py.append(y2);
357 return;
358 }
359 }
360 else
361 {
362 if (d3 < m_distance_tolerance_square)
363 {
364 px.append(x3);
365 py.append(y3);
366 return;
367 }
368 }
369 break;
370 }
371 case 1:
372 {
373 // p1,p2,p4 are collinear, p3 is significant
374 //----------------------
375 if (d3 * d3 <= m_distance_tolerance_square * (dx*dx + dy*dy))
376 {
377 if (m_angle_tolerance < curve_angle_tolerance_epsilon)
378 {
379 px.append(x23);
380 py.append(y23);
381 return;
382 }
383
384 // Angle Condition
385 //----------------------
386 double da1 = fabs(atan2(y4 - y3, x4 - x3) - atan2(y3 - y2, x3 - x2));
387 if (da1 >= M_PI)
388 {
389 da1 = M_2PI - da1;
390 }
391
392 if (da1 < m_angle_tolerance)
393 {
394 px.append(x2);
395 py.append(y2);
396
397 px.append(x3);
398 py.append(y3);
399 return;
400 }
401
402 if (m_cusp_limit > 0.0 || m_cusp_limit < 0.0)
403 {
404 if (da1 > m_cusp_limit)
405 {
406 px.append(x3);
407 py.append(y3);
408 return;
409 }
410 }
411 }
412 break;
413 }
414 case 2:
415 {
416 // p1,p3,p4 are collinear, p2 is significant
417 //----------------------
418 if (d2 * d2 <= m_distance_tolerance_square * (dx*dx + dy*dy))
419 {
420 if (m_angle_tolerance < curve_angle_tolerance_epsilon)
421 {
422 px.append(x23);
423 py.append(y23);
424 return;
425 }
426
427 // Angle Condition
428 //----------------------
429 double da1 = fabs(atan2(y3 - y2, x3 - x2) - atan2(y2 - y1, x2 - x1));
430 if (da1 >= M_PI)
431 {
432 da1 = M_2PI - da1;
433 }
434
435 if (da1 < m_angle_tolerance)
436 {
437 px.append(x2);
438 py.append(y2);
439
440 px.append(x3);
441 py.append(y3);
442 return;
443 }
444
445 if (m_cusp_limit > 0.0 || m_cusp_limit < 0.0)
446 {
447 if (da1 > m_cusp_limit)
448 {
449 px.append(x2);
450 py.append(y2);
451 return;
452 }
453 }
454 }
455 break;
456 }
457 case 3:
458 {
459 // Regular case
460 //-----------------
461 if ((d2 + d3)*(d2 + d3) <= m_distance_tolerance_square * (dx*dx + dy*dy))
462 {
463 // If the curvature doesn't exceed the distance_tolerance value
464 // we tend to finish subdivisions.
465 //----------------------
466 if (m_angle_tolerance < curve_angle_tolerance_epsilon)
467 {
468 px.append(x23);
469 py.append(y23);
470 return;
471 }
472
473 // Angle & Cusp Condition
474 //----------------------
475 const double k = atan2(y3 - y2, x3 - x2);
476 double da1 = fabs(k - atan2(y2 - y1, x2 - x1));
477 double da2 = fabs(atan2(y4 - y3, x4 - x3) - k);
478 if (da1 >= M_PI)
479 {
480 da1 = M_2PI - da1;
481 }
482 if (da2 >= M_PI)
483 {
484 da2 = M_2PI - da2;
485 }
486
487 if (da1 + da2 < m_angle_tolerance)
488 {
489 // Finally we can stop the recursion
490 //----------------------
491
492 px.append(x23);
493 py.append(y23);
494 return;
495 }
496
497 if (m_cusp_limit > 0.0 || m_cusp_limit < 0.0)
498 {
499 if (da1 > m_cusp_limit)
500 {
501 px.append(x2);
502 py.append(y2);
503 return;
504 }
505
506 if (da2 > m_cusp_limit)
507 {
508 px.append(x3);
509 py.append(y3);
510 return;
511 }
512 }
513 }
514 break;
515 }
516 default:
517 break;
518 }
519
520 // Continue subdivision
521 //----------------------
522 PointBezier_r(x1, y1, x12, y12, x123, y123, x1234, y1234, static_cast<qint16>(level + 1), px, py);
523 PointBezier_r(x1234, y1234, x234, y234, x34, y34, x4, y4, static_cast<qint16>(level + 1), px, py);
524}

527/**
528 * @brief GetCubicBezierPoints return list with cubic bezier curve points.
529 * @param p1 first spline point.
530 * @param p2 first control point.
531 * @param p3 second control point.
532 * @param p4 last spline point.
533 * @return list of points.
534 */
535QVector<QPointF> VAbstractCubicBezier::GetCubicBezierPoints(const QPointF &p1, const QPointF &p2, const QPointF &p3,
536 const QPointF &p4)
537{
538 QVector<QPointF> pvector;
539 QVector<qreal> x;
540 QVector<qreal> y;
541 QVector<qreal>& wx = x;
542 QVector<qreal>& wy = y;
543 x.append ( p1.x () );
544 y.append ( p1.y () );
545 PointBezier_r ( p1.x (), p1.y (), p2.x (), p2.y (),
546 p3.x (), p3.y (), p4.x (), p4.y (), 0, wx, wy );
547 x.append ( p4.x () );
548 y.append ( p4.y () );
549 for ( qint32 i = 0; i < x.count(); ++i )
550 {
551 pvector.append( QPointF ( x.at(i), y.at(i)) );
552 }
553 return pvector;
554}

605//---------------------------------------------------------------------------------------------------------------------
606QPair<qreal, qreal> VAbstractCubicBezier::HobbyHandleLengths(const QPointF &p1, const QPointF &p4,
607 qreal angle1Deg, qreal angle2Deg,
608 qreal tensionStart, qreal tensionEnd)
609{
610 const QLineF chord(p1, p4);
611 const qreal d = chord.length();
612
613 if (qFuzzyIsNull(d))
614 {
615 return {0.0, 0.0};
616 }
617
618 const qreal chordRad = qDegreesToRadians(chord.angle());
619 qreal theta = qDegreesToRadians(angle1Deg) - chordRad;
620 // MetaPost convention (mp_set_controls): theta is the departure direction at p1
621 // relative to the chord p1->p4; phi is the chord direction relative to the
622 // arrival direction at p4. Seamly stores the arrival direction as angle2, so
623 // phi = (chordAngle + 180) - angle2, not angle2 - (chordAngle + 180).
624 qreal phi = qDegreesToRadians(chord.angle() + 180.0) - qDegreesToRadians(angle2Deg);
625
626 const qreal twoPi = 2.0 * M_PI;
627 while (theta > M_PI)
628 {
629 theta -= twoPi;
630 }
631 while (theta < -M_PI)
632 {
633 theta += twoPi;
634 }
635 while (phi > M_PI)
636 {
637 phi -= twoPi;
638 }
639 while (phi < -M_PI)
640 {
641 phi += twoPi;
642 }
643
644 const qreal sq2 = qSqrt(2.0);
645 const qreal sqrt5 = qSqrt(5.0);
646 const qreal cA = 0.5 * (sqrt5 - 1.0);
647 const qreal cB = 0.5 * (3.0 - sqrt5);
648
649 auto velocity = [&](qreal a, qreal b) -> qreal
650 {
651 const qreal num = 2.0 + sq2 * (qSin(a) - qSin(b) / 16.0)
652 * (qSin(b) - qSin(a) / 16.0)
653 * (qCos(a) - qCos(b));
654 const qreal den = 3.0 * (1.0 + cA * qCos(a) + cB * qCos(b));
655 return (qAbs(den) > 1e-9) ? num / den : (1.0 / 3.0);
656 };
657
658 // Hobby tension divides the velocity: higher tension -> shorter handles
659 // (curve held tighter to the chord), lower tension -> longer, bulgier handles.
660 const qreal tauStart = (tensionStart > 1e-6) ? tensionStart : 1e-6;
661 const qreal tauEnd = (tensionEnd > 1e-6) ? tensionEnd : 1e-6;
662 const qreal c1 = qBound(1e-4, velocity(theta, phi) * d / tauStart, d * 8.0);
663 const qreal c2 = qBound(1e-4, velocity(phi, theta) * d / tauEnd, d * 8.0);
664
665 return {c1, c2};
666}

668//---------------------------------------------------------------------------------------------------------------------
669// Composite 8-point Gauss-Legendre quadrature over 16 subintervals. A single
670// 8-point panel over [0,1] is not accurate enough for bulgy curves to match
671// LengthBezier (the adaptive-subdivision polyline length used by the tooltip
672// and the VCurveLength formula variable) within the solvers' 0.05mm epsilon;
673// this function must track LengthBezier closely or the solvers converge to a
674// length that then disagrees with what the UI displays.
675qreal VAbstractCubicBezier::CubicBezierLengthGL(const QPointF &p1, const QPointF &p2,
676 const QPointF &p3, const QPointF &p4)
677{
678 static const qreal t[] = {0.01985071506835568, 0.10166676129318664,
679 0.23723379504183550, 0.40828267875217509,
680 0.59171732124782494, 0.76276620495816450,
681 0.89833323870681336, 0.98014928493164430};
682 static const qreal w[] = {0.05061426814518813, 0.11119051722668723,
683 0.15685332293894364, 0.18134189168918099,
684 0.18134189168918099, 0.15685332293894364,
685 0.11119051722668723, 0.05061426814518813};
686 static const int kPanels = 16;
687
688 qreal len = 0.0;
689 for (int panel = 0; panel < kPanels; ++panel)
690 {
691 for (int i = 0; i < 8; ++i)
692 {
693 const qreal s = (panel + t[i]) / kPanels;
694 const qreal q = 1.0 - s;
695 const QPointF d = 3.0 * (p2 - p1) * (q * q)
696 + 6.0 * (p3 - p2) * (q * s)
697 + 3.0 * (p4 - p3) * (s * s);
698 len += w[i] * qSqrt(d.x() * d.x() + d.y() * d.y()) / kPanels;
699 }
700 }
701 return len;
702}

704//---------------------------------------------------------------------------------------------------------------------
705// Bisection solver: find handle lengths (c1, c2) such that the arc length of
706// the cubic Bezier equals targetPx. curveLen(scale) is monotonically
707// increasing in scale, so a bracket [0, hi] found by doubling hi until it
708// overshoots the target is narrowed every iteration -- unlike the previous
709// secant approach, whose out-of-bracket fallback (xn < 0.0 -> x1 * 0.5) did
710// not maintain a bracket at all and could return an unconverged midpoint
711// silently, up to ~27mm off target for unreachable goals.
712//
713// mode: 1=vary start only, 2=vary end only, 3=vary both proportionally
714QPair<qreal, qreal> VAbstractCubicBezier::SolveHandleLengths(
715 const QPointF &p1, const QPointF &p4,
716 qreal angle1Deg, qreal angle2Deg,
717 qreal baseC1, qreal baseC2,
718 qreal targetPx, int mode)
719{
720 const qreal eps = ToPixel(0.05, Unit::Mm);
721
722 if (baseC1 <= 0.0 && baseC2 <= 0.0)
723 {
724 return {baseC1, baseC2};
725 }
726
727 const qreal a1rad = qDegreesToRadians(angle1Deg);
728 const qreal a2rad = qDegreesToRadians(angle2Deg);
729 const qreal cos1 = qCos(a1rad), sin1 = qSin(a1rad);
730 const qreal cos2 = qCos(a2rad), sin2 = qSin(a2rad);
731
732 auto handlesForScale = [&](qreal scale) -> QPair<qreal, qreal>
733 {
734 qreal c1 = baseC1;
735 qreal c2 = baseC2;
736 if (mode == 1)
737 {
738 c1 = baseC1 * scale;
739 }
740 else if (mode == 2)
741 {
742 c2 = baseC2 * scale;
743 }
744 else
745 {
746 c1 = baseC1 * scale;
747 c2 = baseC2 * scale;
748 }
749 return {c1, c2};
750 };
751
752 auto curveLen = [&](qreal scale) -> qreal
753 {
754 const QPair<qreal, qreal> h = handlesForScale(scale);
755 return CubicBezierLengthGL(p1,
756 p1 + QPointF(h.first * cos1, -h.first * sin1),
757 p4 + QPointF(h.second * cos2, -h.second * sin2),
758 p4);
759 };
760
761 const qreal minLen = curveLen(0.0);
762 if (targetPx <= minLen)
763 {
764 return {0.0, 0.0};
765 }
766
767 qreal hi = 1.0;
768 for (int guard = 0; guard < 64 && curveLen(hi) < targetPx; ++guard)
769 {
770 hi *= 2.0;
771 }
772
773 qreal lo = 0.0;
774 for (int i = 0; i < 60; ++i)
775 {
776 const qreal mid = 0.5 * (lo + hi);
777 const qreal f = curveLen(mid) - targetPx;
778 if (qAbs(f) < eps)
779 {
780 return handlesForScale(mid);
781 }
782 if (f < 0.0)
783 {
784 lo = mid;
785 }
786 else
787 {
788 hi = mid;
789 }
790 }
791
792 return handlesForScale(0.5 * (lo + hi));
793}

795//---------------------------------------------------------------------------------------------------------------------
796// Combined auto-smooth + curve-length: instead of scaling the finished Hobby
797// handles linearly, vary the Hobby tension parameter so the curve keeps its
798// natural Hobby shape at the found tension. lenForTau(tau) is monotonically
799// decreasing (higher tension -> shorter handles -> shorter curve), so this is
800// solved by bisection on a fixed bracket [TAU_MIN, TAU_MAX] whose ends cover
801// the whole range reachable through the qBound(1e-4, ..., d*8.0) clamp in
802// HobbyHandleLengths. lo/hi are narrowed every iteration, unlike the previous
803// secant-with-reset approach, whose fallback re-tried the same bracket
804// midpoint forever without ever shrinking it -- causing the second and later
805// evaluations to collapse to a fixed tau and return unconverged results.
806//
807// mode: 1=vary start tension (rho), 2=vary end tension (sigma), 3=both equally
808QPair<qreal, qreal> VAbstractCubicBezier::SolveHobbyTension(
809 const QPointF &p1, const QPointF &p4,
810 qreal angle1Deg, qreal angle2Deg,
811 qreal targetPx, int mode)
812{
813 const qreal eps = ToPixel(0.05, Unit::Mm);
814 const qreal a1rad = qDegreesToRadians(angle1Deg);
815 const qreal a2rad = qDegreesToRadians(angle2Deg);
816 const qreal cos1 = qCos(a1rad), sin1 = qSin(a1rad);
817 const qreal cos2 = qCos(a2rad), sin2 = qSin(a2rad);
818
819 auto handlesForTau = [&](qreal tau) -> QPair<qreal, qreal>
820 {
821 qreal tauStart = 1.0, tauEnd = 1.0;
822 if (mode == 1)
823 {
824 tauStart = tau;
825 }
826 else if (mode == 2)
827 {
828 tauEnd = tau;
829 }
830 else
831 {
832 tauStart = tau;
833 tauEnd = tau;
834 }
835 return HobbyHandleLengths(p1, p4, angle1Deg, angle2Deg, tauStart, tauEnd);
836 };
837
838 auto lenForTau = [&](qreal tau) -> qreal
839 {
840 const QPair<qreal, qreal> h = handlesForTau(tau);
841 const QPointF c1pt = p1 + QPointF(h.first * cos1, -h.first * sin1);
842 const QPointF c2pt = p4 + QPointF(h.second * cos2, -h.second * sin2);
843 return CubicBezierLengthGL(p1, c1pt, c2pt, p4);
844 };
845
846 const qreal tauMin = 1.0 / 1024.0;
847 const qreal tauMax = 1024.0;
848 const qreal lenMax = lenForTau(tauMin);
849 const qreal lenMin = lenForTau(tauMax);
850
851 if (targetPx >= lenMax)
852 {
853 return handlesForTau(tauMin);
854 }
855 if (targetPx <= lenMin)
856 {
857 return handlesForTau(tauMax);
858 }
859
860 qreal lo = tauMin;
861 qreal hi = tauMax;
862 for (int i = 0; i < 60; ++i)
863 {
864 const qreal mid = qSqrt(lo * hi);
865 const qreal f = lenForTau(mid) - targetPx;
866 if (qAbs(f) < eps)
867 {
868 return handlesForTau(mid);
869 }
870 if (f > 0.0)
871 {
872 lo = mid;
873 }
874 else
875 {
876 hi = mid;
877 }
878 }
879
880 return handlesForTau(qSqrt(lo * hi));
881}
