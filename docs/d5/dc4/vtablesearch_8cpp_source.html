77void VTableSearch::find(const QString &text)
78{
79 SCASSERT(m_table != nullptr)
80
81 clear();
82
83 if (!text.isEmpty())
84 {
85 m_searchList = findItems(text);
86
87 if (!m_searchList.isEmpty())
88 {

133void VTableSearch::removeRow(int row)
134{
135 if (m_searchIndex < 0 || m_searchIndex >= m_searchList.size())
136 {
137 return;
138 }
139
140 const int indexRow = m_searchList.at(m_searchIndex)->row();
141
142 if (row <= indexRow)
143 {
144 for (QTableWidgetItem *item : m_searchList)
145 {

186 m_searchList = findItems(text);

241void VTableSearch::clear()
242{
243 SCASSERT(m_table != nullptr)
244
245 for (int i = 0; i < m_table->rowCount(); ++i)
246 {
247 for (int j = 0; j < m_table->columnCount(); ++j)
248 {
249 if (QTableWidgetItem *item = m_table->item(i, j))
250 {
251 if (item->row() % 2 != 0 && m_table->alternatingRowColors())
252 {
253 item->setBackground(QPalette().alternateBase());
254 }
255 else
256 {
257 item->setBackground(QPalette().base());
258 }
259 item->setForeground(QPalette().text());
260 }
261 }
262 }
263
264 m_searchList.clear();
265 m_searchIndex = -1;
266
267 emit hasResult(false);
268}

292QList<QTableWidgetItem *> VTableSearch::findItems(const QString &text)
293{
294 if (m_matchWord && !m_matchCase)
295 {
296 m_table->findItems(text, Qt::MatchFixedString);
297 }
298 else if (m_matchWord && m_matchCase)
299 {
300 m_table->findItems(text, Qt::MatchFixedString | Qt::MatchCaseSensitive);
301 }
302 else if (m_matchRegEx && !m_matchCase)
303 {
304 QRegularExpression regex(text, QRegularExpression::CaseInsensitiveOption);
305 if (regex.isValid())
306 {
307 return m_table->findItems(regex.pattern(), Qt::MatchRegularExpression);
308 }
309 }
310 else if (m_matchRegEx && m_matchCase)
311 {
312 QRegularExpression regex(text);
313 if (regex.isValid())
314 {
315 return m_table->findItems(regex.pattern(), Qt::MatchRegularExpression);
316 }
317 }
318 else if (m_matchCase)
319 {
320 return m_table->findItems(text, Qt::MatchContains | Qt::MatchCaseSensitive);
321 }
322
323 return m_table->findItems(text, Qt::MatchContains);
324}
