158/// @brief DialogAccepted save data and emit signal about closed dialog.
159//---------------------------------------------------------------------------------------------------------------------
160void HistoryDialog::DialogAccepted()
161{
162 QTableWidgetItem *item = ui->tableWidget->item(m_cursorToolRecordRow, IdColumn);
163 quint32 id = qvariant_cast<quint32>(item->data(Qt::UserRole));
164 emit showHistoryTool(id, false);
165
166 // Reset cursor to bottom of table before closing
167 m_doc->setCursorId(NULL_ID);
168
169 emit DialogClosed(QDialog::Accepted);
170}

172//---------------------------------------------------------------------------------------------------------------------
173/// @brief cellClicked changed history record
174/// @param row number row in table
175/// @param column number column in table
176//---------------------------------------------------------------------------------------------------------------------
177void HistoryDialog::cellClicked(int row, int column)
178{
179 if (column == CursorColumn)
180 {
181 // clear current cursor row icon
182 QTableWidgetItem *item = ui->tableWidget->item(m_cursorRow, CursorColumn);
183 item->setIcon(QIcon());
184
185 // set new cursor row icon
186 m_cursorRow = row;
187 item = ui->tableWidget->item(m_cursorRow, CursorColumn);
188 item->setIcon(QIcon("://icon/24x24/left_to_right_arrow.png"));
189
190 // set new cursor position in pattern.
191 // if cursor is at last row set id to 0, else use tool id in row.
192 item = ui->tableWidget->item(m_cursorRow, IdColumn);
193 const quint32 id = qvariant_cast<quint32>(item->data(Qt::UserRole));
194 m_doc->blockSignals(true);
195 row == ui->tableWidget->rowCount()-1 ? m_doc->setCursorId(NULL_ID) : m_doc->setCursorId(id);
196 m_doc->blockSignals(false);
197 }
198 else
199 {
200 // Un Highlight old selected tool in draft block
201 QTableWidgetItem *item = ui->tableWidget->item(m_cursorToolRecordRow, IdColumn);
202 quint32 id = qvariant_cast<quint32>(item->data(Qt::UserRole));
203 emit showHistoryTool(id, false);
204
205 // Highlight new selected tool in draft block
206 m_cursorToolRecordRow = row;
207 item = ui->tableWidget->item(m_cursorToolRecordRow, IdColumn);
208 id = qvariant_cast<quint32>(item->data(Qt::UserRole));
209 emit showHistoryTool(id, true);
210 }
211}

213//---------------------------------------------------------------------------------------------------------------------
214/// @brief cursorChanged changed cursor of input. Cursor show after which record we will insert new object
215/// @param id id of object
216//---------------------------------------------------------------------------------------------------------------------
217void HistoryDialog::cursorChanged(quint32 id)
218{
219 for (qint32 row = 0; row < ui->tableWidget->rowCount(); ++row)
220 {
221 QTableWidgetItem *item = ui->tableWidget->item(row, IdColumn);
222 quint32 rowId = qvariant_cast<quint32>(item->data(Qt::UserRole));
223 if (rowId == id)
224 {
225 QTableWidgetItem *oldCursorItem = ui->tableWidget->item(m_cursorRow, CursorColumn);
226 oldCursorItem->setIcon(QIcon());
227 m_cursorRow = row;
228 item->setIcon(QIcon("://icon/24x24/left_to_right_arrow.png"));
229
230 QScrollBar *verScrollBar = ui->tableWidget->verticalScrollBar();
231 verScrollBar->setSliderPosition(verScrollBar->maximum());
232 }
233 }
234}

249/// @brief fillTable fill table
250//---------------------------------------------------------------------------------------------------------------------
251void HistoryDialog::fillTable()
252{
253 ui->draftblockName_Label->setText(m_doc->getActiveDraftBlockName());
254 ui->tableWidget->clear();
255
256 // Set the description column to stretch
257 QHeaderView *header = ui->tableWidget->horizontalHeader();
258 header->setSectionResizeMode(DescColumn, QHeaderView::Stretch);
259
260 QVector<VToolRecord> history = m_doc->getBlockHistory();
261 qint32 currentRow = -1;
262 qint32 count = 0;
263 ui->tableWidget->setRowCount(history.size());//Set Row count to number of Tool history records
264
265 qint32 size = history.size();
266 for (qint32 i = 0; i < size; ++i)
267 {
268 const VToolRecord tool = history.at(i);
269 const RowData rowData = record(tool);
270 if (!rowData.tool.isEmpty())
271 {
272 currentRow++;
273
274 {
275 QTableWidgetItem *item = new QTableWidgetItem(QString(""));
276 item->setTextAlignment(Qt::AlignHCenter | Qt::AlignVCenter);
277 ui->tableWidget->setItem(currentRow, CursorColumn, item); //1st column is Cursor position
278 }
279
280 {
281 QTableWidgetItem *item = new QTableWidgetItem(QString().setNum(rowData.id));
282 item->setTextAlignment(Qt::AlignHCenter | Qt::AlignVCenter);
283 item->setData(Qt::UserRole, rowData.id);
284 item->setFlags(item->flags() ^ Qt::ItemIsEditable);
285 ui->tableWidget->setItem(currentRow, IdColumn, item); //2nd column is Tool id
286 }
287
288 {
289 QTableWidgetItem *item = new QTableWidgetItem(QString(rowData.name));
290 item->setTextAlignment(Qt::AlignHCenter | Qt::AlignVCenter);
291 item->setFlags(item->flags() ^ Qt::ItemIsEditable);
292 ui->tableWidget->setColumnWidth(NameColumn, 150);
293 ui->tableWidget->setItem(currentRow, NameColumn, item); //3rd column is Tool name or Obj names
294 ui->tableWidget->resizeRowToContents(currentRow);
295 }
296
297 QTableWidgetItem *item = new QTableWidgetItem(rowData.tool);
298 item->setTextAlignment(Qt::AlignLeft | Qt::AlignVCenter);
299 item->setFlags(item->flags() ^ Qt::ItemIsEditable);
300 ui->tableWidget->setColumnWidth(DescColumn, 400);
301 item->setIcon(QIcon(rowData.icon));
302 ui->tableWidget->setItem(currentRow, DescColumn, item); //4th column is Tool description
303
304 {
305 QTableWidgetItem *item = new QTableWidgetItem(QString(rowData.length));
306 item->setTextAlignment(Qt::AlignHCenter | Qt::AlignVCenter);
307 item->setFlags(item->flags() ^ Qt::ItemIsEditable);
308 ui->tableWidget->setColumnWidth(LengthColumn, 120);
309 ui->tableWidget->setItem(currentRow, LengthColumn, item); //5th column is Tool length formula
310 ui->tableWidget->resizeRowToContents(currentRow);
311 }
312
313 {
314 QTableWidgetItem *item = new QTableWidgetItem(QString(rowData.angle));
315 item->setTextAlignment(Qt::AlignHCenter | Qt::AlignVCenter);
316 item->setFlags(item->flags() ^ Qt::ItemIsEditable);
317 ui->tableWidget->setColumnWidth(AngleColumn, 120);
318 ui->tableWidget->setItem(currentRow, AngleColumn, item); //6th column is Tool angle formula
319 ui->tableWidget->resizeRowToContents(currentRow);
320 }
321 ++count;
322 }
323 }
324
325 ui->tableWidget->setRowCount(count);//Current history records count
326
327 if (count>0)
328 {
329 m_cursorRow = cursorRow();
330 QTableWidgetItem *item = ui->tableWidget->item(m_cursorRow, CursorColumn);
331 SCASSERT(item != nullptr)
332 item->setIcon(QIcon("://icon/24x24/left_to_right_arrow.png"));//place arrow in 1st column of most recent history record
333 }
334
335}

345RowData HistoryDialog::record(const VToolRecord &tool)
346{
347 // This check helps to find missed tools in the switch
348 Q_STATIC_ASSERT_X(static_cast<int>(Tool::LAST_ONE_DO_NOT_USE) == 54, "Not all tools were used in history.");

353 const QDomElement domElement = m_doc->elementById(toolId);
354 if (domElement.isElement() == false)
355 {
356 qDebug() << "Can't find element by id" << Q_FUNC_INFO;
357 rowData.icon = ":/icons/win.icon.theme/16x16/status/dialog-warning.png";
358 rowData.name = QString();

361 }
362 try
363 {
364 switch (tool.getTypeTool())
365 {
366 case Tool::Arrow:
367 case Tool::SinglePoint:
368 case Tool::DoublePoint:
369 case Tool::LinePoint:
370 case Tool::AbstractSpline:
371 case Tool::Cut:
372 case Tool::Midpoint:// Same as Tool::AlongLine, but tool will never has such type
373 case Tool::ArcIntersectAxis:// Same as Tool::CurveIntersectAxis, but tool will never has such type

827bool HistoryDialog::eventFilter(QObject *object, QEvent *event)
828{
829 if (QLineEdit *textEdit = qobject_cast<QLineEdit *>(object))

831 if (event->type() == QEvent::KeyPress)
832 {
833 QKeyEvent *keyEvent = static_cast<QKeyEvent *>(event);
834 if ((keyEvent->key() == Qt::Key_Enter) || (keyEvent->key() == Qt::Key_Return))
835 {
836 // Ignore Enter key
837 return true;
838 }
839 else if ((keyEvent->key() == Qt::Key_Period) && (keyEvent->modifiers() & Qt::KeypadModifier))
840 {
841 QString separator = qApp->Settings()->getOsSeparator()
842 ? QString(localeDecimalPoint(QLocale()))
843 : QString(localeDecimalPoint(QLocale::c()));
844
845 textEdit->insert(separator);
846 return true;
847 }
848 }
849 }
850 else
851 {
852 // pass the event on to the parent class
853 return DialogTool::eventFilter(object, event);
854 }
855 return false;// pass the event to the widget
856}

949void HistoryDialog::copyToClipboard()
950{
951 QItemSelectionModel *model = ui->tableWidget->selectionModel();
952 QModelIndexList selectedIndexes = model->selectedIndexes();
953
954 QString clipboardString;
955
956 for (int i = 0; i < selectedIndexes.count(); ++i)
957 {
958 QModelIndex current = selectedIndexes[i];
959 QString displayText = current.data(Qt::DisplayRole).toString();
960
961 // Check if another column exists beyond this one.
962 if (i + 1 < selectedIndexes.count())
963 {
964 QModelIndex next = selectedIndexes[i+1];

966 // If the column is on different row, the clipboard should take note.

968 {
969 displayText.append("\n");
970 }
971 else
972 {
973 // Otherwise append a comma separator.
974 displayText.append(" , ");
975 }
976 }
977 clipboardString.append(displayText);
978 }
979
980 QClipboard *clipboard = QApplication::clipboard();
981 clipboard->setText(clipboardString);
