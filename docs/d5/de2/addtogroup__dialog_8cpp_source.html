125void AddToGroupDialog::addGroup()
126{
127 QScopedPointer<EditGroupDialog> dialog(new EditGroupDialog(new VContainer(qApp->translateVariables(),
128 qApp->patternUnitP()), NULL_ID, this));
129 SCASSERT(dialog != nullptr)
130
131 QString group_name;
132 while (1)
133 {
134 const bool result = dialog->exec();
135 group_name = dialog->getName();
136 if (result == false || group_name.isEmpty())
137 {
138 return;
139 }
140 bool exists = m_doc->groupNameExists(group_name);
141 if (exists == false)
142 {
143 break;
144 }
145
146 QMessageBox message_box;
147 message_box.setWindowTitle(tr("Name Exists"));
148 message_box.setIcon(QMessageBox::Warning);
149 message_box.setStandardButtons(QMessageBox::Retry | QMessageBox::Cancel);
150 message_box.setDefaultButton(QMessageBox::Retry);
151 message_box.setText(tr("The action can't be completed because the group name already exists."));
152 int boxResult = message_box.exec();
153
154 switch (boxResult)
155 {
156 case QMessageBox::Retry:
157 break; // Repeat Add Group Dialog
158 case QMessageBox::Cancel:
159 return; // Exit Add Group Dialog
160 default:
161 break; // should never be reached
162 }
163 }
164
165 const quint32 next_id = VContainer::getNextId();
166
167 const QDomElement group = m_doc->createGroup(next_id, group_name, dialog->getColor(), dialog->getLineType(),
168 dialog->getLineWeight(), dialog->getGroupData());
169
170 if (!group.isNull())
171 {
172 VAbstractMainWindow *window = qobject_cast<VAbstractMainWindow *>(qApp->getMainWindow());
173 SCASSERT(window != nullptr)
174
175 AddGroup *command = new AddGroup(group, m_doc);
176 connect(command, &AddGroup::updateGroups, window, &VAbstractMainWindow::updateGroups);
177 qApp->getUndoStack()->push(command);
178
179 fillNameBox();
180
181 int index = ui->group_name_combobox->findText(group_name);
182 if (index != -1)
183 {
184 ui->group_name_combobox->setCurrentIndex(index);
185 }
186 }
187}
