131//---------------------------------------------------------------------------------------------------------------------
132VToolPointFromArcAndTangent *VToolPointFromArcAndTangent::Create(const quint32 _id,
133 const QString &pointName,
134 quint32 arcId,
135 quint32 tangentPointId,
136 CrossCirclesPoint crossPoint,
137 qreal mx, qreal my,
138 bool showPointName,
139 VMainGraphicsScene *scene,
140 VAbstractPattern *doc,
141 VContainer *data,
142 const Document &parse,
143 const Source &typeCreation)
144{
145 const VArc arc = *data->GeometricObject<VArc>(arcId);
146 const VPointF tPoint = *data->GeometricObject<VPointF>(tangentPointId);
147
148 const QPointF point = VToolPointFromArcAndTangent::FindPoint(static_cast<QPointF>(tPoint), &arc, crossPoint);
149
150 if (point == QPointF())
151 {
152 const QString msg = tr("<b><big>Can't find intersection point %1 of</big></b><br>"
153 "<b><big>%2 and Tangent</big></b><br><br>"
154 "Using origin point as a place holder until pattern is corrected.")
155 .arg(pointName)
156 .arg(arc.name());
157
158 QMessageBox msgBox(qApp->getMainWindow());
159 msgBox.setWindowTitle(tr("Intersect Arc and Tangent"));
160 msgBox.setWindowFlags(msgBox.windowFlags() & ~Qt::WindowContextHelpButtonHint);
161 msgBox.setWindowIcon(QIcon(":/toolicon/32x32/point_from_arc_and_tangent.png"));
162 msgBox.setIcon(QMessageBox::Warning);
163 msgBox.setText(msg);
164 msgBox.setStandardButtons(QMessageBox::Ok);
165 msgBox.exec();
166 }
167
168 quint32 id = _id;
169
170 VPointF *p = new VPointF(point, pointName, mx, my);
171 p->setShowPointName(showPointName);
172
173 if (typeCreation == Source::FromGui)
174 {
175 id = data->AddGObject(p);
176 }
177 else
178 {
179 data->UpdateGObject(id, p);
180 if (parse != Document::FullParse)
181 {
182 doc->UpdateToolData(id, data);
183 }
184 }
185
186 if (parse == Document::FullParse)
187 {
188 VDrawTool::AddRecord(id, Tool::PointFromArcAndTangent, doc);
189 VToolPointFromArcAndTangent *point = new VToolPointFromArcAndTangent(doc, data, id, arcId, tangentPointId,
190 crossPoint, typeCreation);
191
192 scene->addItem(point);
193 InitToolConnections(scene, point);
194 VAbstractPattern::AddTool(id, point);
195 doc->IncrementReferens(arc.getIdTool());
196 doc->IncrementReferens(tPoint.getIdTool());
197 return point;
198 }
199 return nullptr;
200}

203QPointF VToolPointFromArcAndTangent::FindPoint(const QPointF &p, const VArc *arc, const CrossCirclesPoint pType)
204{
205 QPointF p1, p2;
206 const QPointF center = static_cast<QPointF>(arc->GetCenter());
207 const qreal radius = arc->GetRadius();
208 const int res = VGObject::ContactPoints (p, center, radius, p1, p2);
209
210 QLineF r1Arc(center, p1);
211 r1Arc.setLength(radius+10);
212
213 QLineF r2Arc(center, p2);
214 r2Arc.setLength(radius+10);
215
216 switch(res)
217 {
218 case 2:
219 {
220 int localRes = 0;
221 bool flagP1 = false;
222
223 if (arc->IsIntersectLine(r1Arc))
224 {
225 ++localRes;
226 flagP1 = true;
227 }
228
229 if (arc->IsIntersectLine(r2Arc))
230 {
231 ++localRes;
232 }
233
234 switch(localRes)
235 {
236 case 2:
237 if (pType == CrossCirclesPoint::FirstPoint)
238 {
239 return p1;
240 }
241 else
242 {
243 return p2;
244 }
245 case 1:
246 if (flagP1)
247 {
248 return p1;
249 }
250 else
251 {
252 return p2;
253 }
254 case 0:
255 default:
256 return QPointF();
257 }
258
259 break;
260 }
261 case 1:
262 if (arc->IsIntersectLine(r1Arc))
263 {
264 return p1;
265 }
266 else
267 {
268 return QPointF();
269 }
270 case 3:
271 case 0:
272 default:
273 break;
274 }
275 return QPointF();
276}
