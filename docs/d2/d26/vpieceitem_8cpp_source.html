109 /// @brief GetAngle calculates the angle between the line, which goes from rotation center to point and x axis
110 /// @param point point of interest
111 /// @return the angle between line from rotation center and point of interest and x axis
112 //---------------------------------------------------------------------------------------------------------------------
113 double VPieceItem::GetAngle(const QPointF &point) const
114 {
115  const double dX = point.x() - m_rotationCenter.x();
116  const double dY = point.y() - m_rotationCenter.y();
117 
118  if (fabs(dX) < 1 && fabs(dY) < 1)
119  {
120  return 0;
121  }
122  else
123  {
124  return qAtan2(dY, dX);
125  }
126 }
