176/**
177 * @brief ContextMenu show context menu for tool.
178 * @param itemId id of point.
179 * @param event context menu event.
180 * @param showRemove true - tool enable option delete.
181 * @param ref true - do not ignore referens value.
182 */
183void VDrawTool::ContextMenu(QGraphicsSceneContextMenuEvent *event, quint32 itemId,
184 const RemoveOption &showRemove, const Referens &ref)
185{
186 SCASSERT(event != nullptr)
187
188 if (m_suppressContextMenu)
189 {
190 return;
191 }
192
193 GOType itemType = GOType::Unknown;
194 if(itemId != NULL_ID)
195 {
196 try
197 {
198 itemType = data.GetGObject(itemId)->getType();
199 }
200 catch (const VExceptionBadId &error)
201 { // Possible case. Parent was deleted, but the node object is still here.
202 qWarning() << qUtf8Printable(error.ErrorMessage());
203 }
204 }
205
206 quint32 toolId = this->getId();
207 QMap<quint32, Tool> history = doc->getGroupObjHistory();
208 Tool tooltype = history.value(toolId);
209
210 qCDebug(vTool, "Creating tool context menu.");
211 QMenu menu;
212
213 // Actions for copying tool length and angle
214 QAction *actionCopyToolLength = nullptr;
215 QAction *actionCopyLineAngle = nullptr;
216
217 QAction *actionOption = menu.addAction(QIcon::fromTheme("preferences-other"), VDrawTool::tr("Properties"));
218
219 // Show object name menu item
220 QAction *actionShowPointName = menu.addAction(QIcon("://icon/16x16/open_eye.png"), VDrawTool::tr("Show Point Name"));
221 actionShowPointName->setCheckable(true);
222
223 if (itemType == GOType::Point)
224 {
225 actionShowPointName->setChecked(isPointNameVisible(itemId));
226 }
227 else
228 {
229 actionShowPointName->setVisible(false);
230 }
231
232 // Add Copy menu
233 QMenu *copyMenu = menu.addMenu(QIcon("://icon/32x32/clipboard_icon.png"), VDrawTool::tr("Copy"));
234 actionCopyToolLength = copyMenu->addAction(QCoreApplication::translate("VDrawTool", "Length"));
235 actionCopyLineAngle = copyMenu->addAction(QCoreApplication::translate("VDrawTool", "Angle"));
236
237 // Only add the Angle submneu for the point tools that add a line.
238 if (tooltype != Tool::Line &&
239 tooltype != Tool::EndLine &&
240 tooltype != Tool::Bisector &&
241 tooltype != Tool::Height &&
242 tooltype != Tool::Normal &&
243 tooltype != Tool::LineIntersectAxis &&
244 tooltype != Tool::AlongLine &&
245 tooltype != Tool::CurveIntersectAxis)
246 {
247 actionCopyLineAngle->setVisible(false);
248 }
249
250 QAction *actionDelete = menu.addAction(QIcon::fromTheme("edit-delete"), VDrawTool::tr("Delete"));
251 if (showRemove == RemoveOption::Enable)
252 {
253 if (ref == Referens::Follow)
254 {
255 if (isUsed())
256 {
257 qCDebug(vTool, "Delete disabled. Tool has children.");
258 actionDelete->setEnabled(false);
259 }
260 else
261 {
262 qCDebug(vTool, "Delete enabled. Tool has no children.");
263 actionDelete->setEnabled(true);
264 }
265 }
266 else
267 {
268 qCDebug(vTool, "Delete enabled. Ignore referens value.");
269 actionDelete->setEnabled(true);
270 }
271 }
272 else
273 {
274 qCDebug(vTool, "Delete disabled.");
275 actionDelete->setEnabled(false);
276 }
277
278 QMap<quint32,QString> groupsContainingItem = doc->getGroupsContainingItem(toolId, itemId, true);
279 QMap<quint32,QString> groupsNotContainingItem = doc->getGroupsContainingItem(toolId, itemId, false);
280
281 // Add Move Group Item menu
282 QActionGroup *actionMoveGroupMenu= new QActionGroup(this);
283
284 if (!groupsContainingItem.empty() && !groupsNotContainingItem.empty())
285 {
286 QMenu *menuMoveGroupItem = menu.addMenu(QIcon("://icon/svg/list-move.svg"), VDrawTool::tr("Move Group Object"));
287 QStringList sourceList = QStringList(groupsContainingItem.values());
288 QStringList destList = QStringList(groupsNotContainingItem.values());
289 sourceList.sort(Qt::CaseInsensitive);
290 destList.sort(Qt::CaseInsensitive);
291
292 for(int i=0; i < sourceList.count(); ++i)
293 {
294 const quint32 sourecId = groupsContainingItem.key(sourceList[i]);
295
296 for(int j=0; j < destList.count(); ++j)
297 {
298 QAction *actionMoveGroupItem = menuMoveGroupItem->addAction(VDrawTool::tr("From ") + sourceList[i] + VDrawTool::tr(" to ") + destList[j]);
299 actionMoveGroupMenu->addAction(actionMoveGroupItem);
300 const quint32 destId = groupsNotContainingItem.key(destList[j]);
301
302 QPair<quint32, quint32> data(sourecId, destId);
303 QVariant variantData = QVariant::fromValue(data);
304 actionMoveGroupItem->setData(variantData);
305 }
306 }
307 }
308
309 // Add Group Item menu item
310 QActionGroup *actionAddGroupMenu= new QActionGroup(this);
311 if (!groupsNotContainingItem.empty())
312 {
313 QMenu *menuAddGroupItem = menu.addMenu(QIcon("://icon/32x32/add.png"), VDrawTool::tr("Add Group Object"));
314 QStringList list = QStringList(groupsNotContainingItem.values());
315 list.sort(Qt::CaseInsensitive);
316
317 for(int i=0; i<list.count(); ++i)
318 {
319 QAction *actionAddGroupItem = menuAddGroupItem->addAction(list[i]);
320 actionAddGroupMenu->addAction(actionAddGroupItem);
321 const quint32 groupId = groupsNotContainingItem.key(list[i]);
322 actionAddGroupItem->setData(groupId);
323 groupsNotContainingItem.remove(groupId); // delete any duplicate groups
324 }
325 }
326
327
328
329 // Remove Group Item menu item
330 QActionGroup *actionDeleteGroupMenu = new QActionGroup(this);
331
332 if (!groupsContainingItem.empty())
333 {
334 QMenu *menuRemoveGroupItem = menu.addMenu(QIcon("://icon/32x32/remove.png"), VDrawTool::tr("Remove Group Object"));
335
336 QStringList list = QStringList(groupsContainingItem.values());
337 list.sort(Qt::CaseInsensitive);
338
339 for(int i=0; i<list.count(); ++i)
340 {
341 QAction *actionRemoveGroupItem = menuRemoveGroupItem->addAction(list[i]);
342 actionDeleteGroupMenu->addAction(actionRemoveGroupItem);
343 const quint32 groupId = groupsContainingItem.key(list[i]);
344 actionRemoveGroupItem->setData(groupId);
345 groupsContainingItem.remove(groupId);
346 }
347 }
348
349 QAction *selectedAction = menu.exec(event->screenPos());
350 if(selectedAction == nullptr)
351 {
352 return;
353 }
354 if (selectedAction == actionOption)
355 {
356 qCDebug(vTool, "Show options.");
357 qApp->getSceneView()->itemClicked(nullptr);
358 m_dialog = QSharedPointer<Dialog>(new Dialog(getData(), m_id, qApp->getMainWindow()));
359 m_dialog->setModal(true);
360
361 connect(m_dialog.data(), &DialogTool::DialogClosed, this, &VDrawTool::FullUpdateFromGuiOk);
362 connect(m_dialog.data(), &DialogTool::DialogApplied, this, &VDrawTool::FullUpdateFromGuiApply);
363
364 this->setDialog();
365
366 m_dialog->show();
367 }
368 else if (selectedAction == actionDelete)
369 {
370 qCDebug(vTool, "Deleting Object.");
371 deleteTool(); // do not catch exception here
372 return; //Leave this method immediately after call!!!
373 }
374 else if (selectedAction == actionShowPointName)
375 {
376 updatePointNameVisibility(itemId, selectedAction->isChecked());
377 }
378
379 else if (selectedAction == actionCopyToolLength)
380 {
381 QString text = QString("");
382
383 switch (tooltype)
384 {
385 case Tool::Line:
386 {
387 const QDomElement domElement = doc->elementById(toolId);
388 if (domElement.isElement())
389 {
390 text = VDrawTool::tr("Line_") +
391 data.GeometricObject<VPointF>(doc->GetParametrUInt(domElement, AttrFirstPoint, "0"))->name() +
392 "_"+
393 data.GeometricObject<VPointF>(doc->GetParametrUInt(domElement, AttrSecondPoint, "0"))->name();
394 break;
395 }
396 }
397 case Tool::Arc:
398 case Tool::ArcWithLength:
399 {
400 const QSharedPointer<VArc> arc = data.GeometricObject<VArc>(toolId);
401 SCASSERT(!arc.isNull())
402 text = arc->NameForHistory(VDrawTool::tr("Arc_"));
403 break;
404 }
405 case Tool::ShoulderPoint:
406 case Tool::Normal:
407 case Tool::Bisector:
408 case Tool::LineIntersect:
409 case Tool::BasePoint:
410 case Tool::EndLine:
411 case Tool::PointOfContact:
412 case Tool::Height:
413 case Tool::Triangle:
414 case Tool::PointOfIntersection:
415 case Tool::CutArc:
416 case Tool::CutSpline:
417 case Tool::CutSplinePath:
418 case Tool::LineIntersectAxis:
419 case Tool::CurveIntersectAxis:
420 case Tool::PointOfIntersectionArcs:
421 case Tool::PointOfIntersectionCircles:
422 case Tool::PointOfIntersectionCurves:
423 case Tool::PointFromCircleAndTangent:
424 case Tool::PointFromArcAndTangent:
425 {
426 const QSharedPointer<VGObject> obj = data.GetGObject(toolId);
427 SCASSERT(!obj.isNull())
428 text = obj->name();
429 break;
430 }
431
432 case Tool::EllipticalArc:
433 {
434 const QSharedPointer<VEllipticalArc> elArc = data.GeometricObject<VEllipticalArc>(toolId);
435 SCASSERT(!elArc.isNull())
436 text = elArc->NameForHistory(VDrawTool::tr("ElArc_"));
437 break;
438 }
439
440 case Tool::Spline:
441 {
442 const QSharedPointer<VSpline> spl = data.GeometricObject<VSpline>(toolId);
443 SCASSERT(!spl.isNull())
444 text = spl->NameForHistory(VDrawTool::tr("Spl_"));
445 break;
446 }
447
448 case Tool::SplinePath:
449 {
450 const QSharedPointer<VSplinePath> splPath = data.GeometricObject<VSplinePath>(toolId);
451 SCASSERT(!splPath.isNull())
452 text = splPath->NameForHistory(VDrawTool::tr("SplPath_"));
453 break;
454 }
455
456 case Tool::CubicBezier:
457 {
458 const QSharedPointer<VCubicBezier> spl = data.GeometricObject<VCubicBezier>(toolId);
459 SCASSERT(!spl.isNull())
460 text = spl->NameForHistory(VDrawTool::tr("Spl_"));
461 break;
462 }
463
464 case Tool::CubicBezierPath:
465 {
466 const QSharedPointer<VCubicBezierPath> splPath = data.GeometricObject<VCubicBezierPath>(toolId);
467 SCASSERT(!splPath.isNull())
468 text = splPath->NameForHistory(VDrawTool::tr("SplPath_"));
469 break;
470 }
471
472 case Tool::Rotation:
473 case Tool::Move:
474 case Tool::MirrorByLine:
475 case Tool::MirrorByAxis:
476 {
477 const QSharedPointer<VGObject> obj = data.GetGObject(itemId);
478 SCASSERT(!obj.isNull())
479 text = obj->name();
480 break;
481 }
482 }
483
484 QClipboard *clipboard = QApplication::clipboard();
485 clipboard->setText(text);
486 }
487 else if (selectedAction->actionGroup() == actionAddGroupMenu)
488 {
489 quint32 groupId = selectedAction->data().toUInt();
490 QDomElement item = doc->addGroupItem(this->getId(), itemId, groupId);
491
492 VMainGraphicsScene *scene = qobject_cast<VMainGraphicsScene *>(qApp->getCurrentScene());
493 SCASSERT(scene != nullptr)
494 scene->clearSelection();

497 SCASSERT(window != nullptr)

501 qApp->getUndoStack()->push(command);

503 }
504 else if (selectedAction->actionGroup() == actionMoveGroupMenu)
505 {
506 QVariant retrievedVariant = selectedAction->data();
507 QPair<quint32, quint32> retrievedPair = retrievedVariant.value<QPair<quint32, quint32>>();

530 }
531 }
