73/// @brief Class showing a widget as popup window.
74///
75/// @param parent
76///
77/// @details
78/// - If parent not specified (default), then popup widget gets
79/// attribute Qt::WA_DeleteOnClose and will be deleted after close.
80/// - setParent() function allows you to specify the widget to be popped up.
81/// After widget is set, you normally should call show() slot in order to pop the
82/// widget up at the specified global position.
83/// - WidgetPopup takes care about positioning of your widget on the screen so it will
84/// be always visible even if popped beside.
85//---------------------------------------------------------------------------------------------------------------------
86WidgetPopup::WidgetPopup(QWidget *parent)
87 : QFrame(parent, Qt::Popup)
88 , m_parent(nullptr)
89 , m_owned(true)
90 , m_oldParent(nullptr)
91 , m_duration(-1)
92{
93 setAttribute(Qt::WA_WindowPropagation);
94
95 if (parentWidget() == nullptr)
96 {
97 setAttribute(Qt::WA_DeleteOnClose);
98 }
99
100 setLayout(new QVBoxLayout());
101 layout()->setContentsMargins(10, 10, 10, 10);
102}

105/// @brief setParent Set popup parent
106///
107/// This method sets the parent of the popup.
108///
109/// @param parent
110/// @param owned
111///
112/// @details
113/// - If owned is true then the widget will be reparented to the popup widget.
114//---------------------------------------------------------------------------------------------------------------------
115void WidgetPopup::setParent(QWidget *parent, bool owned)
116{
117 if (m_parent)
118 {
119 layout()->removeWidget(m_parent);
120
121 if (m_owned)
122 {
123 m_parent->setParent(nullptr);
124 delete m_parent;
125 }
126 else
127 {
128 m_parent->setParent(m_oldParent);
129 }
130 }
131
132 m_parent = parent;
133 m_owned = owned;
134 m_oldParent = nullptr;
135
136 if (m_parent)
137 {
138 m_oldParent = m_parent->parentWidget();
139 m_parent->setParent(this);
140 layout()->addWidget(m_parent);
141 }
142}

145/// @brief popupMessage Creat popup message
146///
147/// This methaod createa popup message of a given duratio in ms.
148///
149/// @param parent
150/// @param msg
151/// @param duration
152//---------------------------------------------------------------------------------------------------------------------
153void WidgetPopup::popupMessage(const QString &msg, const int &duration, QWidget *parent)
154{
155 SCASSERT(parent != nullptr)
156
157 WidgetPopup *popup = new WidgetPopup(parent);
158 QLabel *label = new QLabel(msg);
159 QFont font = label->font();
160 font.setBold(true);
161 font.setPixelSize(16);
162 label->setFont(font);
163 popup->setParent(label);
164 popup->setDuration(duration);
165 popup->show(parent->frameGeometry().center());
166}

197
