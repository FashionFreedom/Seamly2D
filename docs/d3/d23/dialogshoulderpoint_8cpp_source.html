207/**
208 * @brief ChoosedObject gets id and type of selected object. Save right data and ignore wrong.
209 * @param id id of point or detail
210 * @param type type of object
211 */
212void DialogShoulderPoint::ChosenObject(quint32 id, const SceneObject &type)
213{
214 if (prepare == false)
215 {
216 if (type == SceneObject::Point)
217 {
218 VisToolShoulderPoint *line = qobject_cast<VisToolShoulderPoint *>(vis);
219 SCASSERT(line != nullptr)
220
221 switch (number)
222 {
223 case 0:
224 if (SetObject(id, ui->comboBoxP3, tr("Select first point of line")))
225 {
226 number++;
227 line->VisualMode(id);
228 }
229 break;
230 case 1:
231 if (getCurrentObjectId(ui->comboBoxP3) != id)
232 {
233 if (SetObject(id, ui->comboBoxP1Line, tr("Select second point of line")))
234 {
235 number++;
236 line->setLineP1Id(id);
237 line->RefreshGeometry();
238 }
239 }
240 break;
241 case 2:
242 {
243 QSet<quint32> set;
244 set.insert(getCurrentObjectId(ui->comboBoxP3));
245 set.insert(getCurrentObjectId(ui->comboBoxP1Line));
246 set.insert(id);
247
248 if (set.size() == 3)
249 {
250 if (SetObject(id, ui->comboBoxP2Line, ""))
251 {
252 line->setLineP2Id(id);
253 line->RefreshGeometry();
254 prepare = true;
255 this->setModal(true);
256 this->show();
257 }
258 }
259 }
260 break;
261 default:
262 break;
263 }
264 }
265 }
266}

269void DialogShoulderPoint::SaveData()
270{
271 pointName = ui->lineEditNamePoint->text();
272 formula = ui->plainTextEditFormula->toPlainText();
273 formula.replace("\n", " ");
274
275 VisToolShoulderPoint *line = qobject_cast<VisToolShoulderPoint *>(vis);
276 SCASSERT(line != nullptr)
277
278 line->setObject1Id(GetP3());
279 line->setLineP1Id(GetP1Line());
280 line->setLineP2Id(GetP2Line());
281 line->setLength(formula);
282 line->setLineStyle(lineTypeToPenStyle(getLineType()));
283 line->setLineWeight(getLineWeight());
284 line->RefreshGeometry();
285}
