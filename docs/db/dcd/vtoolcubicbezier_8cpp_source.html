81//---------------------------------------------------------------------------------------------------------------------
82// Set the spline's control points from handle lengths (c1, c2) along the given
83// directions, preserving the original canvas point IDs.
84static void setHandlesFromLengths(VCubicBezier *spline, const QPointF &p1pt, const QPointF &p4pt,
85 qreal angle1, qreal angle2, qreal c1, qreal c2,
86 quint32 p2Id, quint32 p3Id)
87{
88 QLineF newH1(p1pt, p1pt + QPointF(c1, 0.0));
89 newH1.setAngle(angle1);
90 QLineF newH2(p4pt, p4pt + QPointF(c2, 0.0));
91 newH2.setAngle(angle2);
92
93 const VPointF oldP2 = spline->GetP2();
94 const VPointF oldP3 = spline->GetP3();
95 VPointF newP2(newH1.p2(), oldP2.name(), oldP2.mx(), oldP2.my(),
96 oldP2.getIdObject(), oldP2.getMode());
97 newP2.setId(p2Id);
98 VPointF newP3(newH2.p2(), oldP3.name(), oldP3.mx(), oldP3.my(),
99 oldP3.getIdObject(), oldP3.getMode());
100 newP3.setId(p3Id);
101
102 spline->SetP2(newP2);
103 spline->SetP3(newP3);
104}

169//---------------------------------------------------------------------------------------------------------------------
170VToolCubicBezier *VToolCubicBezier::Create(const quint32 _id, VCubicBezier *spline,
171 bool autoSmooth, int lengthMode,
172 const QString &targetLength,
173 VMainGraphicsScene *scene,
174 VAbstractPattern *doc, VContainer *data,
175 const Document &parse, const Source &typeCreation)
176{
177 const quint32 origP2Id = spline->GetP2().id();
178 const quint32 origP3Id = spline->GetP3().id();
179
180 // Mutual-exclusive dispatch: the three handle-computation modes (Hobby,
181 // length-solver, combined tension-solver) each produce a DIFFERENT set of
182 // handle lengths from the SAME inputs. Running them sequentially (Hobby
183 // first, then solver on the result) would feed solver inputs that are
184 // already transformed, producing wrong geometry. The if/else-if ensures
185 // exactly one code-path modifies the handles per Create() call.
186 {
187 const QPointF p1pt = static_cast<QPointF>(spline->GetP1());
188 const QPointF p4pt = static_cast<QPointF>(spline->GetP4());
189 const QLineF h1(p1pt, static_cast<QPointF>(spline->GetP2()));
190 const QLineF h2(p4pt, static_cast<QPointF>(spline->GetP3()));
191 const qreal angle1 = h1.angle();
192 const qreal angle2 = h2.angle();
193
194 const bool hasTarget = (lengthMode > 0 && !targetLength.isEmpty());
195 qreal targetPx = 0.0;
196 if (hasTarget)
197 {
198 QString tl = targetLength;
199 targetPx = qApp->toPixel(CheckFormula(_id, tl, data));
200 }
201
202 qreal c1 = h1.length();
203 qreal c2 = h2.length();
204 bool modified = false;
205
206 if (autoSmooth && hasTarget && targetPx > 0.0)
207 {
208 // Combined: vary Hobby tension to hit the target length.
209 const QPair<qreal, qreal> s = VAbstractCubicBezier::SolveHobbyTension(
210 p1pt, p4pt, angle1, angle2, targetPx, lengthMode);
211 c1 = s.first;
212 c2 = s.second;
213 modified = true;
214 }
215 else if (autoSmooth)
216 {
217 const QPair<qreal, qreal> s = VAbstractCubicBezier::HobbyHandleLengths(
218 p1pt, p4pt, angle1, angle2);
219 c1 = s.first;
220 c2 = s.second;
221 modified = true;
222 }
223 else if (hasTarget && targetPx > 0.0)
224 {
225 const QPair<qreal, qreal> s = VAbstractCubicBezier::SolveHandleLengths(
226 p1pt, p4pt, angle1, angle2, h1.length(), h2.length(), targetPx, lengthMode);
227 c1 = s.first;
228 c2 = s.second;
229 modified = true;
230 }
231
232 if (modified)
233 {
234 setHandlesFromLengths(spline, p1pt, p4pt, angle1, angle2, c1, c2, origP2Id, origP3Id);
235 }
236 }
237
238 quint32 id = _id;
239 if (typeCreation == Source::FromGui)
240 {
241 id = data->AddGObject(spline);
242 data->AddSpline(data->GeometricObject<VAbstractBezier>(id), id);
243 }
244 else
245 {
246 data->UpdateGObject(id, spline);
247 data->AddSpline(data->GeometricObject<VAbstractBezier>(id), id);
248 if (parse != Document::FullParse)
249 {
250 doc->UpdateToolData(id, data);
251 }
252 }
253
254 if (parse == Document::FullParse)
255 {
256 VDrawTool::AddRecord(id, Tool::CubicBezier, doc);
257 auto _spl = new VToolCubicBezier(doc, data, id, autoSmooth, lengthMode, targetLength, typeCreation);
258 scene->addItem(_spl);
259 initSplineToolConnections(scene, _spl);
260 VAbstractPattern::AddTool(id, _spl);
261 doc->IncrementReferens(spline->GetP1().getIdTool());
262 doc->IncrementReferens(spline->GetP2().getIdTool());
263 doc->IncrementReferens(spline->GetP3().getIdTool());
264 doc->IncrementReferens(spline->GetP4().getIdTool());
265 return _spl;
266 }
267 return nullptr;
268}

464void VToolCubicBezier::SetSplineAttributes(QDomElement &domElement, const VCubicBezier &spl)
465{
466 SCASSERT(doc != nullptr)
467
468 doc->SetAttribute(domElement, AttrType, ToolType);
469 doc->SetAttribute(domElement, AttrPoint1, spl.GetP1().id());
470 doc->SetAttribute(domElement, AttrPoint2, m_p2Id);
471 doc->SetAttribute(domElement, AttrPoint3, m_p3Id);
472 doc->SetAttribute(domElement, AttrPoint4, spl.GetP4().id());
473
474 if (spl.GetDuplicate() > 0)
475 {
476 doc->SetAttribute(domElement, AttrDuplicate, spl.GetDuplicate());
477 }
478 else
479 {
480 if (domElement.hasAttribute(AttrDuplicate))
481 {
482 domElement.removeAttribute(AttrDuplicate);
483 }
484 }
485
486 if (m_autoSmooth)
487 {
488 doc->SetAttribute(domElement, AttrAutoSmooth, QStringLiteral("true"));
489 }
490 else
491 {
492 domElement.removeAttribute(AttrAutoSmooth);
493 }
494
495 if (m_lengthMode > 0)
496 {
497 doc->SetAttribute(domElement, AttrLengthMode, m_lengthMode);
498 }
499 else
500 {
501 domElement.removeAttribute(AttrLengthMode);
502 }
503
504 // Persist the target length independently of the mode so toggling the mode
505 // Off and back On does not lose the user's entered value.
506 if (!m_targetLength.isEmpty())
507 {
508 doc->SetAttribute(domElement, AttrLength, m_targetLength);
509 }
510 else
511 {
512 domElement.removeAttribute(AttrLength);
513 }
