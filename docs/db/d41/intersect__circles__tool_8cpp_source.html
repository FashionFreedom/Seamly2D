140//---------------------------------------------------------------------------------------------------------------------
141IntersectCirclesTool *IntersectCirclesTool::Create(const quint32 _id, const QString &pointName,
142 quint32 firstCircleCenterId,
143 quint32 secondCircleCenterId,
144 QString &firstCircleRadius,
145 QString &secondCircleRadius,
146 CrossCirclesPoint crossPoint,
147 qreal mx, qreal my,
148 bool showPointName,
149 VMainGraphicsScene *scene,
150 VAbstractPattern *doc, VContainer *data,
151 const Document &parse,
152 const Source &typeCreation)
153{
154 const qreal calcC1Radius = qApp->toPixel(CheckFormula(_id, firstCircleRadius, data));
155 const qreal calcC2Radius = qApp->toPixel(CheckFormula(_id, secondCircleRadius, data));
156
157 const VPointF c1Point = *data->GeometricObject<VPointF>(firstCircleCenterId);
158 const VPointF c2Point = *data->GeometricObject<VPointF>(secondCircleCenterId);
159
160 const QPointF point = FindPoint(static_cast<QPointF>(c1Point), static_cast<QPointF>(c2Point), calcC1Radius,
161 calcC2Radius, crossPoint);
162 if (point == QPointF())
163 {
164 const QString msg = tr("<b><big>Can't find intersection point %1 of Circles</big></b><br>"
165 "Using origin point as a place holder until pattern is corrected.")
166 .arg(pointName);
167
168 QMessageBox msgBox(qApp->getMainWindow());
169 msgBox.setWindowTitle(tr("Point Intersect Circles"));
170 msgBox.setWindowFlags(msgBox.windowFlags() & ~Qt::WindowContextHelpButtonHint);
171 msgBox.setWindowIcon(QIcon(":/toolicon/32x32/point_of_intersection_circles.png"));
172 msgBox.setIcon(QMessageBox::Warning);
173 msgBox.setText(msg);
174 msgBox.setStandardButtons(QMessageBox::Ok);
175 msgBox.exec();
176 }
177
178 quint32 id = _id;
179
180 VPointF *p = new VPointF(point, pointName, mx, my);
181 p->setShowPointName(showPointName);
182
183 if (typeCreation == Source::FromGui)
184 {
185 id = data->AddGObject(p);
186 }
187 else
188 {
189 data->UpdateGObject(id, p);
190 if (parse != Document::FullParse)
191 {
192 doc->UpdateToolData(id, data);
193 }
194 }
195
196 if (parse == Document::FullParse)
197 {
198 VDrawTool::AddRecord(id, Tool::PointOfIntersectionCircles, doc);
199 IntersectCirclesTool *point = new IntersectCirclesTool(doc, data, id, firstCircleCenterId,
200 secondCircleCenterId,
201 firstCircleRadius,
202 secondCircleRadius, crossPoint,
203 typeCreation);
204 scene->addItem(point);
205 InitToolConnections(scene, point);
206 VAbstractPattern::AddTool(id, point);
207 doc->IncrementReferens(c1Point.getIdTool());
208 doc->IncrementReferens(c2Point.getIdTool());
209 return point;
210 }
211 return nullptr;
212}

214//---------------------------------------------------------------------------------------------------------------------
215QPointF IntersectCirclesTool::FindPoint(const QPointF &c1Point, const QPointF &c2Point, qreal c1Radius,
216 qreal c2Radius, const CrossCirclesPoint crossPoint)
217{
218 QPointF p1, p2;
219 const int res = VGObject::IntersectionCircles(c1Point, c1Radius, c2Point, c2Radius, p1, p2);
220
221 switch(res)
222 {
223 case 2:
224 if (crossPoint == CrossCirclesPoint::FirstPoint)
225 {
226 return p1;
227 }
228 else
229 {
230 return p2;
231 }
232 case 1:
233 return p1;
234 case 3:
235 case 0:
236 default:
237 return QPointF();
238 }
239}
