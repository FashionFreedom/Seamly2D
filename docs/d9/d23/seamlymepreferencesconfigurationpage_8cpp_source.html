199void SeamlyMePreferencesConfigurationPage::Apply()
200{
201 VSeamlyMeSettings *settings = qApp->seamlyMeSettings();
202
203 settings->setShowWelcome(ui->showWelcome_CheckBox->isChecked());
204 if (ui->cLocale_RadioButton->isChecked())
205 {
206 settings->setOsSeparator(false);
207 }
208 else
209 {
210 settings->setOsSeparator(true);
211 }
212 settings->setToolBarStyle(ui->toolBarStyle_CheckBox->isChecked());
213
214 settings->setUseNativeDialogs(ui->use_native_checkox->isChecked());
215
216 if (m_themeChanged)
217 {
218 int theme = qvariant_cast<int>(ui->theme_ComboBox->currentData());
219 settings->setAppTheme(theme);
220 qApp->setTheme();
221 m_themeChanged = false;
222 }
223
224 if (m_langChanged || m_systemChanged)
225 {
226 const QString locale = qvariant_cast<QString>(ui->langCombo->currentData());
227 settings->setLocale(locale);
228 m_langChanged = false;
229
230 const QString code = qvariant_cast<QString>(ui->systemCombo->currentData());
231 settings->SetPMSystemCode(code);
232 m_systemChanged = false;
233
234 qApp->loadTranslations(locale);
235 qApp->processEvents();// force to call changeEvent
236
237 // Part about measurments will not be updated automatically
238 qApp->retranslateTables();
239 qApp->retranslateGroups();
240 }
241
242 if (m_defGradationChanged)
243 {
244 settings->SetDefHeight(ui->defHeightCombo->currentText().toInt());
245 settings->SetDefSize(ui->defSizeCombo->currentText().toInt());
246 m_defGradationChanged = false;
247 }
248}
