329// @brief Configures a raw QComboBox to act as a locked, filtered font selector.
330// @param combo_box The target font QComboBox to populate.
331// @param label The companion preview QLabel to style.
332// @param initial_font The initial saved QFont setting.
333// @param size_combo_box Optional companion size QComboBox (pass nullptr if not used).
334// @param fixed_size Optional explicit font size override (set to 0 to use dynamic sizing).
335//---------------------------------------------------------------------------------------------------------------------
336void PreferencesGraphicsViewPage::setupFontComboBox(QComboBox *combo_box, QLabel *label, const QFont &initial_font, QComboBox *size_combo_box, int fixed_size)
337{
338 if (!combo_box || !label)
339 {
340 return;
341 }
342
343 // Configure stable horizontal layout policies
344 combo_box->setSizeAdjustPolicy(QComboBox::AdjustToContentsOnFirstShow);
345 combo_box->setEnabled(true);
346
347 // Populate and filter fonts safely
348 QFontDatabase font_database;
349 QStringList font_families = font_database.families();
350 int fallback_size = initial_font.pointSize();
351
352 for (int i = 0; i < font_families.count(); ++i)
353 {
354 QString font_name = font_families.at(i);
355
356 // Enforce vector outlines across all platforms
357 if (font_database.isSmoothlyScalable(font_name))
358 {
359 // Defensive cross-platform check to drop explicit bitmap containers
360 if (font_name.contains("bitmap", Qt::CaseInsensitive) ||
361 font_name.contains("GB18030", Qt::CaseInsensitive))
362 {
363 continue;
364 }
365
366 combo_box->addItem(font_name);
367 int inserted_index = combo_box->count() - 1;
368
369 // Set individual dropdown row preview text style (fixed to 10pt for clean scannability)
370 QFont row_font(font_name, 10);
371 combo_box->setItemData(inserted_index, row_font, Qt::FontRole);
372 }
373 }
374
375 // Set up initial current font choices
376 int font_index = combo_box->findText(initial_font.family());
377 if (font_index >= 0)
378 {
379 combo_box->setCurrentIndex(font_index);
380 }
381
382 // Enforce standardized point size for the main layout boxes to avoid layout shifting
383 QFont base_font = initial_font;
384 base_font.setPointSize(10);
385 combo_box->setFont(base_font);
386
387 // Style the preview label to match the baseline font setting
388 int initial_target_size = getTargetSize(size_combo_box, fixed_size, fallback_size);
389 QFont preview_font = initial_font;
390 preview_font.setPointSize(initial_target_size);
391 label->setFont(preview_font);
392
393 // Handle runtime font family changes
394 connect(combo_box, &QComboBox::currentTextChanged, label, [this, combo_box, label,
395 size_combo_box, fixed_size, fallback_size](const QString &family)
396 {
397 // Adjust the closed combo box to prevent visual size bouncing
398 QFont closed_box_font(family);
399 closed_box_font.setPointSize(10);
400 combo_box->setFont(closed_box_font);
401
402 // Adjust the companion preview text label style using the class method
403 int dynamic_size = getTargetSize(size_combo_box, fixed_size, fallback_size);
404 QFont display_font(family);
405 display_font.setPointSize(dynamic_size);
406 label->setFont(display_font);
407 });
408
409 // Handle runtime font size changes (if a companion size box exists)
410 if (size_combo_box)
411 {
412 connect(size_combo_box, &QComboBox::currentTextChanged, label, [this, label, size_combo_box,
413 fixed_size, fallback_size]()
414 {
415 int dynamic_size = getTargetSize(size_combo_box, fixed_size, fallback_size);
416 QFont label_font = label->font();
417 label_font.setPointSize(dynamic_size);
418 label->setFont(label_font);
419 });
420 }
421
422 // Lock visual box vertical layout dimensions perfectly
423 combo_box->ensurePolished();
424 label->ensurePolished();
425
426 //combo_box->setFixedHeight(22);
427}
