67/// @brief childElements serialize the child elements of a tool element.
68///
69/// Some tools keep object references in child elements instead of attributes, for example the points of a
70/// spline path. Text nodes are skipped.
71///
72/// @param element tool element.
73/// @return the child elements as one string.
74//---------------------------------------------------------------------------------------------------------------------
75
76QString childElements(const QDomElement &element)
77{
78 QString dump;
79 QTextStream stream(&dump);
80
81 QDomElement child = element.firstChildElement();
82 while (!child.isNull())
83 {
84 child.save(stream, 0);
85 child = child.nextSiblingElement();
86 }
87 return dump;
88}

91/// @brief referencesChanged check if a tool option change replaced a referenced object.
92///
93/// The reference counters that guard deletion are only set when a tool is created during a full parse.
94/// When an option change replaces a referenced object the counters become stale, so the caller has to
95/// request a full parse to rebuild them. See issue #1521.
96///
97/// @param oldXml tool element before the change.
98/// @param newXml tool element after the change.
99/// @return true if an object reference changed.
100//---------------------------------------------------------------------------------------------------------------------
101
102bool referencesChanged(const QDomElement &oldXml, const QDomElement &newXml)
103{
104 // Attributes of the draw tools that store the id of another object.
105 static const QStringList reference_attributes = QStringList()
106 << AttrBasePoint << AttrFirstPoint << AttrSecondPoint << AttrThirdPoint
107 << AttrCenter << AttrCCenter << AttrC1Center << AttrC2Center
108 << AttrArc << AttrFirstArc << AttrSecondArc
109 << AttrCurve << AttrCurve1 << AttrCurve2
110 << AttrPoint1 << AttrPoint2 << AttrPoint3 << AttrPoint4
111 << AttrP1Line << AttrP2Line << AttrP1Line1 << AttrP2Line1 << AttrP1Line2 << AttrP2Line2
112 << AttrAxisP1 << AttrAxisP2 << AttrTangent << AttrPShoulder
113 << AttrDartP1 << AttrDartP2 << AttrDartP3
114 << AttrBaseLineP1 << AttrBaseLineP2;
115
116 for (int i = 0; i < reference_attributes.size(); ++i)
117 {
118 if (oldXml.attribute(reference_attributes.at(i)) != newXml.attribute(reference_attributes.at(i)))
119 {
120 return true;
121 }
122 }
123
124 return childElements(oldXml) != childElements(newXml);
125}

142void SaveToolOptions::undo()
143{
144 qCDebug(vUndo, "Undo.");
145
146 QDomElement domElement = doc->elementById(nodeId);
147 if (domElement.isElement())
148 {
149 domElement.parentNode().replaceChild(oldXml, domElement);
150
151 if (referencesChanged(oldXml, newXml))
152 {
153 // A referenced object changed. Only a full parse rebuilds the reference counters. See issue #1521.
154 emit NeedFullParsing();
155 }
156 else
157 {
158 emit NeedLiteParsing(Document::LiteParse);
159 }
160 }
161 else
162 {
163 qCWarning(vUndo, "Can't find tool with id = %u.", nodeId);
164 return;
165 }
166}

196bool SaveToolOptions::mergeWith(const QUndoCommand *command)
197{
198 const SaveToolOptions *saveCommand = static_cast<const SaveToolOptions *>(command);
199 SCASSERT(saveCommand != nullptr)
200 const quint32 id = saveCommand->getToolId();
201
202 if (id != nodeId)
203 {
204 return false;
205 }
206
207 newXml = saveCommand->getNewXml();
208 return true;
209}
