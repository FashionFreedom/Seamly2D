77static QChar GetChar(const QString &formula, int &index)
78{
79 if (index >= formula.size())
80 {
81 return QmuEOF;
82 }
83
84 return formula.at(index++);
85}

88static QChar EatWhiteSpace(const QString &formula, int &index)
89{
90 QChar c;
91 do
92 {
93 c = GetChar(formula, index);
94 }
95 while ( c != QmuEOF && c.isSpace() );
96
97 return c;
98}

101static int CheckChar(QChar &c, const QLocale &locale, const QChar &decimal, const QChar &thousand)
102{
103 INIT_LOCALE_VARIABLES(locale);
104 Q_UNUSED(decimalPoint)
105 Q_UNUSED(groupSeparator)
106
107 if (c == positiveSign)
108 {
109 c = '+';
110 return InputToken::InputSign;
111 }
112 else if (c == negativeSign)
113 {
114 c = '-';
115 return InputToken::InputSign;
116 }
117 else if (c == char0)
118 {
119 c = '0';
120 return InputToken::InputDigit;
121 }
122 else if (c == char1)
123 {
124 c = '1';
125 return InputToken::InputDigit;
126 }
127 else if (c == char2)
128 {
129 c = '2';
130 return InputToken::InputDigit;
131 }
132 else if (c == char3)
133 {
134 c = '3';
135 return InputToken::InputDigit;
136 }
137 else if (c == char4)
138 {
139 c = '4';
140 return InputToken::InputDigit;
141 }
142 else if (c == char5)
143 {
144 c = '5';
145 return InputToken::InputDigit;
146 }
147 else if (c == char6)
148 {
149 c = '6';
150 return InputToken::InputDigit;
151 }
152 else if (c == char7)
153 {
154 c = '7';
155 return InputToken::InputDigit;
156 }
157 else if (c == char8)
158 {
159 c = '8';
160 return InputToken::InputDigit;
161 }
162 else if (c == char9)
163 {
164 c = '9';
165 return InputToken::InputDigit;
166 }
167 else if (c == decimal)
168 {
169 return InputToken::InputDot;
170 }
171 else if (c == thousand)
172 {
173 return InputToken::InputThousand;
174 }
175 else if (c == expLower)
176 {
177 c = 'e';
178 return InputToken::InputExp;
179 }
180 else if (c == expUpper)
181 {
182 c = 'E';
183 return InputToken::InputExp;
184 }
185 else
186 {
187 return 0;
188 }
189
190 return 0;
191}

194int ReadVal(const QString &formula, qreal &val, const QLocale &locale, const QChar &decimal, const QChar &thousand)
195{
196 // Must not be equal
197 if (decimal == thousand || formula.isEmpty())
198 {
199 val = 0;
200 return -1;
201 }
202
203 INIT_LOCALE_VARIABLES(locale);
204 Q_UNUSED(decimalPoint)
205 Q_UNUSED(groupSeparator)
206
207 QSet<QChar> reserved;
208 reserved << positiveSign
209 << negativeSign
210 << char0
211 << char1
212 << char2
213 << char3
214 << char4
215 << char5
216 << char6
217 << char7
218 << char8
219 << char9
220 << expUpper
221 << expLower;
222
223 if (reserved.contains(decimal) || reserved.contains(thousand))
224 {
225 val = 0;
226 return -1;
227 }
228
229 // row - current state, column - new state
230 static uchar table[9][6] =
231 {
232 /* None InputSign InputThousand InputDigit InputDot InputExp */
233 { 0, State::Sign, 0, State::Mantissa, State::Dot, 0, }, // Init
234 { 0, 0, 0, State::Mantissa, State::Dot, 0, }, // Sign
235 { 0, 0, 0, State::Mantissa, 0, 0, }, // Thousand
236 { State::Done, State::Done, State::Thousand, State::Mantissa, State::Dot, State::ExpMark,}, // Mantissa
237 { 0, 0, 0, State::Abscissa, 0, 0, }, // Dot
238 { State::Done, State::Done, 0, State::Abscissa, 0, State::ExpMark,}, // Abscissa
239 { 0, State::ExpSign, 0, State::Exponent, 0, 0, }, // ExpMark
240 { 0, 0, 0, State::Exponent, 0, 0, }, // ExpSign
241 { State::Done, 0, 0, State::Exponent, 0, State::Done } // Exponent
242 };
243
244 int state = State::Init; // parse state
245 QString buf;
246
247 int index = 0; // start position
248 QChar c = EatWhiteSpace(formula, index);
249
250 while ( true )
251 {
252 const int input = CheckChar(c, locale, decimal, thousand);// input token
253
254 state = table[state][input];
255
256 if (state == 0)
257 {
258 val = 0;
259 return -1;
260 }
261 else if (state == Done)
262 {
263 // Convert to C locale
264 QLocale cLocale(QLocale::C);
265 const QChar cDecimal = localeDecimalPoint(cLocale);
266 const QChar cThousand = localeGroupSeparator(cLocale);
267 if (locale != cLocale && (cDecimal != decimal || cThousand != thousand))
268 {
269 if (decimal == cThousand)
270 {// Handle reverse to C locale case: thousand '.', decimal ','
271 //const QChar tmpThousand = '@';
272 const QChar tmpThousand = QLatin1Char('@');
273 buf.replace(thousand, tmpThousand);
274 buf.replace(decimal, cDecimal);
275 buf.replace(tmpThousand, cThousand);
276 }
277 else
278 {
279 buf.replace(thousand, cThousand);
280 buf.replace(decimal, cDecimal);
281 }
282 }
283
284 bool ok = false;
285 const double d = cLocale.toDouble(buf, &ok);
286 if (ok)
287 {
288 val = d;
289 return buf.size();
290 }
291 else
292 {
293 val = 0;
294 return -1;
295 }
296 }
297
298 buf.append(c);
299 c = GetChar(formula, index);
300 }
301
302 return -1;
303}

306QString NameRegExp()
307{
308 static QString regex;
309
310 if (regex.isEmpty())
311 {
312 const QList<QLocale> allLocales = QLocale::matchingLocales(QLocale::AnyLanguage,
313 QLocale::AnyScript,
314 QLocale::AnyCountry);
315
316 QString positiveSigns;
317 QString negativeSigns;
318 QString decimalPoints;
319 QString groupSeparators;
320 for(int i = 0; i < allLocales.size(); ++i)
321 {
322 if (!positiveSigns.contains(localePositiveSign(allLocales.at(i))))
323 {
324 positiveSigns.append(localePositiveSign(allLocales.at(i)));
325 }
326
327 if (!negativeSigns.contains(localeNegativeSign(allLocales.at(i))))
328 {
329 negativeSigns.append(localeNegativeSign(allLocales.at(i)));
330 }
331
332 if (!decimalPoints.contains(localeDecimalPoint(allLocales.at(i))))
333 {
334 decimalPoints.append(localeDecimalPoint(allLocales.at(i)));
335 }
336
337 if (!groupSeparators.contains(localeGroupSeparator(allLocales.at(i))))
338 {
339 groupSeparators.append(localeGroupSeparator(allLocales.at(i)));
340 }
341 }
342
343 negativeSigns.replace('-', "\\-");
344 groupSeparators.remove('\'');
345
346 //Same regexp in pattern.xsd schema file. Don't forget to synchronize.
347 // \p{Nd} - \p{Decimal_Digit_Number}
348 // \p{Zs} - \p{Space_Separator}
349 regex = QString("^([^\\p{Nd}\\p{Zs}*/&|!<>^\\()%1%2%3%4=?:;'\"]){1,1}"
350 "([^\\p{Zs}*/&|!<>^\\()%1%2%3%4=?:;\"]){0,}$")
351 .arg(negativeSigns).arg(positiveSigns).arg(decimalPoints).arg(groupSeparators);
352 }
353
354 return regex;
355}

359{
360 return locale.name() != QStringLiteral("tok_001") &&

362 locale.negativeSign().size() == 1 &&
