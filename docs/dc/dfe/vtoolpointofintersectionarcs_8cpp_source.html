128//---------------------------------------------------------------------------------------------------------------------
129VToolPointOfIntersectionArcs *VToolPointOfIntersectionArcs::Create(const quint32 _id, const QString &pointName,
130 const quint32 &firstArcId,
131 const quint32 &secondArcId, CrossCirclesPoint pType,
132 qreal mx, qreal my,
133 VMainGraphicsScene *scene, VAbstractPattern *doc,
134 VContainer *data, const Document &parse,
135 const Source &typeCreation)
136{
137 const QSharedPointer<VArc> firstArc = data->GeometricObject<VArc>(firstArcId);
138 const QSharedPointer<VArc> secondArc = data->GeometricObject<VArc>(secondArcId);
139
140 const QPointF point = FindPoint(firstArc.data(), secondArc.data(), pType);
141
142 if (point == QPointF())
143 {
144 const QString msg = tr("<b><big>Can't find intersection point %1 of Arcs</big></b><br>"
145 "Using origin point as a place holder until pattern is corrected.")
146 .arg(pointName);
147
148 QMessageBox msgBox(qApp->getMainWindow());
149 msgBox.setWindowTitle(tr("Point Intersect Arcs"));
150 msgBox.setWindowFlags(msgBox.windowFlags() & ~Qt::WindowContextHelpButtonHint);
151 msgBox.setWindowIcon(QIcon(":/toolicon/32x32/point_of_intersection_arcs.png"));
152 msgBox.setIcon(QMessageBox::Warning);
153 msgBox.setText(msg);
154 msgBox.setStandardButtons(QMessageBox::Ok);
155 msgBox.exec();
156 }
157
158 quint32 id = _id;
159 if (typeCreation == Source::FromGui)
160 {
161 id = data->AddGObject(new VPointF(point, pointName, mx, my));
162 }
163 else
164 {
165 data->UpdateGObject(id, new VPointF(point, pointName, mx, my));
166 if (parse != Document::FullParse)
167 {
168 doc->UpdateToolData(id, data);
169 }
170 }
171
172 if (parse == Document::FullParse)
173 {
174 VDrawTool::AddRecord(id, Tool::PointOfIntersectionArcs, doc);
175 VToolPointOfIntersectionArcs *point = new VToolPointOfIntersectionArcs(doc, data, id, firstArcId,
176 secondArcId, pType, typeCreation);
177 scene->addItem(point);
178 InitToolConnections(scene, point);
179 VAbstractPattern::AddTool(id, point);
180 doc->IncrementReferens(firstArc->getIdTool());
181 doc->IncrementReferens(secondArc->getIdTool());
182 return point;
183 }
184 return nullptr;
185}

188QPointF VToolPointOfIntersectionArcs::FindPoint(const VArc *arc1, const VArc *arc2, const CrossCirclesPoint pType)
189{
190 QPointF p1, p2;
191 const QPointF centerArc1 = static_cast<QPointF>(arc1->GetCenter());
192 const QPointF centerArc2 = static_cast<QPointF>(arc2->GetCenter());
193 const int res = VGObject::IntersectionCircles(centerArc1, arc1->GetRadius(), centerArc2, arc2->GetRadius(), p1, p2);
194
195 QLineF r1Arc1(centerArc1, p1);
196 r1Arc1.setLength(r1Arc1.length()+10);
197
198 QLineF r1Arc2(centerArc2, p1);
199 r1Arc2.setLength(r1Arc2.length()+10);
200
201 QLineF r2Arc1(centerArc1, p2);
202 r2Arc1.setLength(r2Arc1.length()+10);
203
204 QLineF r2Arc2(centerArc2, p2);
205 r2Arc2.setLength(r2Arc2.length()+10);
206
207 switch(res)
208 {
209 case 2:
210 {
211 int localRes = 0;
212 bool flagP1 = false;
213
214 if (arc1->IsIntersectLine(r1Arc1) && arc2->IsIntersectLine(r1Arc2))
215 {
216 ++localRes;
217 flagP1 = true;
218 }
219
220 if (arc1->IsIntersectLine(r2Arc1) && arc2->IsIntersectLine(r2Arc2))
221 {
222 ++localRes;
223 }
224
225 switch(localRes)
226 {
227 case 2:
228 if (pType == CrossCirclesPoint::FirstPoint)
229 {
230 return p1;
231 }
232 else
233 {
234 return p2;
235 }
236 case 1:
237 if (flagP1)
238 {
239 return p1;
240 }
241 else
242 {
243 return p2;
244 }
245 case 0:
246 default:
247 return QPointF();
248 }
249
250 break;
251 }
252 case 1:
253 if (arc1->IsIntersectLine(r1Arc1) && arc2->IsIntersectLine(r1Arc2))
254 {
255 return p1;
256 }
257 else
258 {
259 return QPointF();
260 }
261 case 3:
262 case 0:
263 default:
264 break;
265 }
266 return QPointF();
267}
