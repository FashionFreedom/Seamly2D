50 /**
51  * @brief Merge all the SVGs in the m_domList list into a single SVG
52  * @return The merged SVG as a DOM document
53  * @details m_domList contains DOM representations of multiple SVGs
54  Assuming each svg contains a main group containing every graphical item of the svg,
55  this function adds to the first svg of the list all the main groups of the other svgs,
56  thus creating a single svg with each svg of the list in it, every svg being in its own group.
57  This function is used in order to create svgs containing groups
58  */
59 QDomDocument SvgGenerator::mergeSvgDoms()
60 {
61  if (m_domList.isEmpty()) {
62  qDebug() << "Error : the SVG list is empty";
63  return QDomDocument();
64  }
65 
66  QDomDocument mergedSvg = m_domList.at(0).cloneNode().toDocument();
67 
68  QDomElement mergedSvgRoot = mergedSvg.documentElement();
69  if (mergedSvgRoot.tagName() != "svg") {
70  qDebug() << "Error : the first SVG does not contain a <svg> tag.";
71  return QDomDocument();
72  }
73 
74  QDomNodeList mergedSvgGroups = mergedSvgRoot.elementsByTagName("g");
75  if (mergedSvgGroups.isEmpty()) {
76  qDebug() << "Error : the SVG does not contain a <g> tag.";
77  return QDomDocument();
78  }
79  mergedSvgRoot.removeChild(mergedSvgGroups.at(0));
80 
81  for (int i = 0; i < m_domList.size(); ++i) {
82  QDomDocument domSvg = m_domList.at(i);
83  QDomElement svgRoot = domSvg.documentElement();
84  if (svgRoot.tagName() != "svg") {
85  qDebug() << "Error : the SVG does not contain a <svg> tag.";
86  return QDomDocument();
87  }
88  QDomNodeList svgGroups = svgRoot.elementsByTagName("g");
89  if (svgGroups.isEmpty()) {
90  qDebug() << "Error : the SVG does not contain a <g> tag.";
91  return QDomDocument();
92  }
93  QDomElement mainGroup = svgGroups.at(0).toElement();
94  cleanSvg(mainGroup);
95  mergedSvgRoot.appendChild(mainGroup);
96  }
97 
98  return mergedSvg;
99 }

102 /**
103  * @brief Remove empty groups from the SVG
104  * @return void
105  * @details This function removes empty unuseful groups from the exported SVG.
106  * Those empty groups were generated by Qt svg generator.
107  */
108 void SvgGenerator::removeEmptyGroups(QDomElement &mainGroup)
109 {
110  QDomNodeList groups = mainGroup.elementsByTagName("g");
111  for (int i = 0; i < groups.size(); ++i) {
112  QDomElement group = groups.at(i).toElement();
113  if (group.childNodes().isEmpty()) {
114  if (mainGroup.removeChild(group).isNull()) {
115  qDebug() << "Error : could not remove empty group";
116  }
117  }
118  }
119 }

122 /**
123  * @brief Remove the M0,0 origin path from the SVG
124  * @return void
125  * @details This function removes tht M0,0 origin path from the SVG.
126  * If this path is not removed, the bounding box of the exported pattern piece
127  * when opened in a svg editing software can be wrong.
128  */
129 void SvgGenerator::removeEmptyOriginPath(QDomElement &mainGroup)
130 {
131  QDomNodeList paths = mainGroup.elementsByTagName("path");
132  for (int i = 0; i < paths.size(); ++i) {
133  QDomElement path = paths.at(i).toElement();
134  if (path.attribute("d") == "M0,0") {
135  QDomElement parentGroup = path.parentNode().toElement();
136  parentGroup.removeChild(path);
137  mainGroup.removeChild(parentGroup);
138  break;
139  }
140  }
141 }

157 /**
158  * @brief Add a new SVG to the list of SVGs to be merged into a single SVG
159  * @param scene : the scene that must be converted to SVG
160  * @return void
161  * @details This function creates a SVG from the given scene and converts it into
162  a DOM that is added to the m_domList list of SVGs to be merged.
163 */
164 void SvgGenerator::addSvgFromScene(QGraphicsScene *scene)
165 {
166  QByteArray byteArray;
167  QBuffer buffer(&byteArray);
168  buffer.open(QIODevice::WriteOnly);
169 
170  QSvgGenerator svgGenerator;
171  svgGenerator.setOutputDevice(&buffer);
172  svgGenerator.setSize(m_paper->rect().size().toSize());
173  svgGenerator.setViewBox(m_paper->rect());
174  svgGenerator.setTitle(QObject::tr("Pattern"));
175  svgGenerator.setResolution(m_resolution);
176 
177  QPainter painter;
178  painter.begin(&svgGenerator);
179  painter.setFont( QFont( "Arial", 8, QFont::Normal ) );
180  painter.setRenderHint(QPainter::Antialiasing, true);
181  painter.setBrush ( QBrush ( Qt::NoBrush ) );
182  scene->render(&painter, m_paper->rect(), m_paper->rect(), Qt::IgnoreAspectRatio);
183  painter.end();
184 
185  QDomDocument domDoc;
186  if (domDoc.setContent(byteArray)) {
187  m_domList.append(domDoc);
188  } else {
189  qDebug() << "Error : Impossible to load the SVG content in the QDomDocument.";
190  }
191 
192  buffer.close();
193 }
