144PatternPieceTool *PatternPieceTool::Create(quint32 id, VPiece newPiece, QString &width, VMainGraphicsScene *scene,
145 VAbstractPattern *doc, VContainer *data, const Document &parse,
146 const Source &typeCreation, const QString &blockName)
147{
148 if (typeCreation == Source::FromGui || typeCreation == Source::FromTool)
149 {
150 data->AddVariable(currentSeamAllowance, new CustomVariable(data, currentSeamAllowance, 0, newPiece.GetSAWidth(),
151 width, true, tr("Current seam allowance")));
152 id = data->AddPiece(newPiece);
153 }
154 else
155 {
156 const qreal calcWidth = CheckFormula(id, width, data);
157 newPiece.setSeamAllowanceWidthFormula(width, calcWidth);
158
159 data->AddVariable(currentSeamAllowance, new CustomVariable(data, currentSeamAllowance, 0, calcWidth,
160 width, true, tr("Current seam allowance")));
161
162 data->UpdatePiece(id, newPiece);
163 if (parse != Document::FullParse)
164 {
165 doc->UpdateToolData(id, data);
166 }
167 }
168
169 PatternPieceTool *patternPiece = nullptr;
170 if (parse == Document::FullParse)
171 {
172 VAbstractTool::AddRecord(id, Tool::Piece, doc);
173 patternPiece = new PatternPieceTool(doc, data, id, typeCreation, scene, blockName);
174 scene->addItem(patternPiece);
175 connect(patternPiece, &PatternPieceTool::chosenTool, scene, &VMainGraphicsScene::chosenItem);
176 connect(scene, &VMainGraphicsScene::EnableDetailItemHover, patternPiece, &PatternPieceTool::AllowHover);
177 connect(scene, &VMainGraphicsScene::EnableDetailItemSelection, patternPiece, &PatternPieceTool::AllowSelecting);
178 connect(scene, &VMainGraphicsScene::highlightPiece, patternPiece, &PatternPieceTool::highlight);
179 connect(scene, &VMainGraphicsScene::pieceLockedChanged, patternPiece, &PatternPieceTool::pieceLockedChanged);
180
181 VAbstractPattern::AddTool(id, patternPiece);
182 patternPiece->RefreshGeometry(); // Refresh internal paths
183 }
184 //Very important to delete it. Only this tool need this special variable.
185 data->RemoveVariable(currentSeamAllowance);
186 return patternPiece;
187}

204void PatternPieceTool::insertNodes(const QVector<VPieceNode> &nodes, quint32 pieceId, VMainGraphicsScene *scene,
205 VContainer *data, VAbstractPattern *doc)
206{
207 SCASSERT(scene != nullptr)
208 SCASSERT(data != nullptr)
209 SCASSERT(doc != nullptr)
210
211 if (pieceId > NULL_ID && !nodes.isEmpty())
212 {
213 VPiece oldPiece;
214 try
215 {
216 oldPiece = data->GetPiece(pieceId);
217 }
218 catch (const VExceptionBadId &)
219 {
220 return;
221 }
222
223 QVector<VPieceNode> newNodes = removeDuplicateNodePoints(oldPiece, nodes, data);
224
225 VPiece newPiece = oldPiece;
226 for (auto node : newNodes)
227 {
228 const quint32 id = PrepareNode(node, scene, doc, data);
229 if (id == NULL_ID)
230 {
231 return;
232 }
233
234 node.SetId(id);
235 newPiece.GetPath().Append(node);
236
237 // Seam allowance tool already initializated and can't init the node
238 PatternPieceTool *patternPiece = qobject_cast<PatternPieceTool*>(VAbstractPattern::getTool(pieceId));
239 SCASSERT(patternPiece != nullptr);
240
241 initializeNode(node, scene, data, doc, patternPiece);
242 }
243
244 SavePieceOptions *saveCommand = new SavePieceOptions(oldPiece, newPiece, doc, pieceId);
245 qApp->getUndoStack()->push(saveCommand);// First push then make a connect
246
247 data->UpdatePiece(pieceId, newPiece);// Update piece because first save will not call lite update
248 connect(saveCommand, &SavePieceOptions::NeedLiteParsing, doc, &VAbstractPattern::LiteParseTree);
249 }
250}

1015QVariant PatternPieceTool::itemChange(QGraphicsItem::GraphicsItemChange change, const QVariant &value)
1016{
1017 if (change == ItemPositionChange && scene())
1018 {
1019 // Each time we move something we call recalculation scene rect. In some cases this can cause moving
1020 // objects positions. And this cause infinite redrawing. That's why we wait the finish of saving the last move.
1021 static bool changeFinished = true;
1022 if (changeFinished)
1023 {
1024 changeFinished = false;
1025
1026 // value - this is new position.
1027 const QPointF newPos = value.toPointF();
1028
1029 MovePiece *cmd = new MovePiece(doc, newPos.x(), newPos.y(), m_id, scene());
1030 //connect(moveDet, &MovePiece::NeedLiteParsing, doc, &VAbstractPattern::LiteParseTree);
1031 qApp->getUndoStack()->push(cmd);
1032
1033 const QList<QGraphicsView *> viewList = scene()->views();
1034 if (!viewList.isEmpty())
1035 {
1036 if (VMainGraphicsView *view = qobject_cast<VMainGraphicsView *>(viewList.at(0)))
1037 {
1038 const qreal scale = sceneScale(scene());
1039 const QRectF viewRect = VMainGraphicsView::SceneVisibleArea(view);
1040 const QRectF itemRect = mapToScene(boundingRect()|childrenBoundingRect()).boundingRect();
1041
1042 // Ensure visible only small rect around a cursor
1043 VMainGraphicsScene *currentScene = qobject_cast<VMainGraphicsScene *>(scene());
1044 SCASSERT(currentScene);
1045 const QPointF cursorPosition = currentScene->getScenePos();
1046 view->ensureVisible(QRectF(cursorPosition.x()-5/scale, cursorPosition.y()-5/scale,
1047 10/scale, 10/scale));
1048 }
1049 }
1050
1051 // Don't forget to update geometry, because first change never call full parse
1052 VPiece piece = VAbstractTool::data.GetPiece(m_id);
1053 piece.SetMx(newPos.x());
1054 piece.SetMy(newPos.y());
1055 VAbstractTool::data.UpdatePiece(m_id, piece);
1056
1057 RefreshGeometry();
1058
1059 changeFinished = true;
1060 }
1061 }
1062
1063 if (change == QGraphicsItem::ItemSelectedChange)
1064 {
1065 if (value == true)
1066 {
1067 // do stuff if selected
1068 this->setFocus();
1069 }
1070 else
1071 {
1072 // do stuff if not selected
1073 }
1074 }
1075
1076 return QGraphicsPathItem::itemChange(change, value);
1077}

1864VPieceItem::MoveTypes PatternPieceTool::findGrainlineGeometry(const VGrainlineData &data, qreal &length,
1865 qreal &rotationAngle, qreal &arrowLength, QPointF &pos)
1866{
1867 const quint32 topAnchorPoint = data.topAnchorPoint();
1868 const quint32 bottomAnchorPoint = data.bottomAnchorPoint();
1869
1870 if (topAnchorPoint != NULL_ID && bottomAnchorPoint != NULL_ID)
1871 {
1872 try
1873 {
1874 const auto topAnchor_Point = VAbstractTool::data.GeometricObject<VPointF>(topAnchorPoint);
1875 const auto bottomAnchor_Point = VAbstractTool::data.GeometricObject<VPointF>(bottomAnchorPoint);
1876
1877 QLineF grainline(static_cast<QPointF>(*bottomAnchor_Point), static_cast<QPointF>(*topAnchor_Point));
1878 length = FromPixel(grainline.length(), *VDataTool::data.GetPatternUnit());
1879 rotationAngle = grainline.angle();
1880
1881 Calculator cal3;
1882 arrowLength = cal3.EvalFormula(VAbstractTool::data.DataVariables(), data.getArrowLength());
1883
1884 // override arrow length formula to ensure that the arrow length does not excede
1885 // 1/2 the length of the grainline length when using the top and bottom anchor points.
1886 if (arrowLength > length / 2)
1887 {
1888 arrowLength = length / 2.1;
1889 }
1890
1891 if (!VFuzzyComparePossibleNulls(rotationAngle, 0))
1892 {
1893 grainline.setAngle(0);
1894 }
1895
1896 pos = grainline.p1();
1897
1898 return VPieceItem::NotMovable;
1899 }
1900 catch(const VExceptionBadId &)
1901 {
1902 // do nothing.
1903 }
1904 }
1905
1906 VPieceItem::MoveTypes restrictions = VPieceItem::AllModifications;
1907 try
1908 {
1909 if (!qmu::QmuTokenParser::IsSingle(data.getRotation()))
1910 {
1911 restrictions &= ~ VPieceItem::IsRotatable;
1912 }
1913
1914 Calculator cal1;
1915 rotationAngle = cal1.EvalFormula(VAbstractTool::data.DataVariables(), data.getRotation());
1916
1917 if (!qmu::QmuTokenParser::IsSingle(data.getLength()))
1918 {
1919 restrictions &= ~ VPieceItem::IsResizable;
1920 }
1921
1922 Calculator cal2;
1923 length = cal2.EvalFormula(VAbstractTool::data.DataVariables(), data.getLength());
1924
1925 Calculator cal3;
1926 arrowLength = cal3.EvalFormula(VAbstractTool::data.DataVariables(), data.getArrowLength());
1927 }
1928 catch(qmu::QmuParserError &error)
1929 {
1930 Q_UNUSED(error);
1931 return VPieceItem::Error;
1932 }
1933
1934 const quint32 centerAnchor = data.centerAnchorPoint();
1935 if (centerAnchor != NULL_ID)
1936 {
1937 try
1938 {
1939 const auto centerAnchorPoint = VAbstractTool::data.GeometricObject<VPointF>(centerAnchor);
1940
1941 const qreal cLength = ToPixel(length, *VDataTool::data.GetPatternUnit());
1942 QLineF grainline(centerAnchorPoint->x(), centerAnchorPoint->y(),
1943 centerAnchorPoint->x() + cLength / 2.0, centerAnchorPoint->y());
1944
1945 grainline.setAngle(rotationAngle);
1946 grainline = QLineF(grainline.p2(), grainline.p1());
1947 grainline.setLength(cLength);
1948
1949 pos = grainline.p2();
1950 restrictions &= ~ VPieceItem::IsMovable;
1951 }
1952 catch(const VExceptionBadId &)
1953 {
1954 pos = data.GetPos();
1955 }
1956 }
1957 else
1958 {
1959 pos = data.GetPos();
1960 }
1961
1962 return restrictions;
1963}
