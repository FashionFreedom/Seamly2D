209void FinalMeasurementsDialog::filterMeasurements(const QString &filterString)
210{
211 for (int row = 0; row < ui->measurements_TableWidget->rowCount(); ++row)
212 {
213 bool match = filterString.isEmpty();
214 for (int column = 0; column < ui->measurements_TableWidget->columnCount() && !match; ++column)
215 {
216 const QTableWidgetItem *item = ui->measurements_TableWidget->item(row, column);
217 match = item != nullptr && item->text().contains(filterString, Qt::CaseInsensitive);
218 }
219 ui->measurements_TableWidget->setRowHidden(row, !match);
220 }

394//---------------------------------------------------------------------------------------------------------------------
395/// @brief evaluationData returns a copy of the data of the last tool in the pattern history, so formulas have access
396/// to every object of the pattern: line lengths and angles, curve lengths and angles, custom variables, measurements,
397/// etc. Falls back to @p data when the history has no tool.
398//---------------------------------------------------------------------------------------------------------------------
399VContainer FinalMeasurementsDialog::evaluationData(const VContainer *data, VPattern *doc)
400{
401 VContainer evalData(*data);
402 const QVector<VToolRecord> *history = doc->getHistory();
403 for (int i = history->size() - 1; i >= 0; --i)
404 {
405 try
406 {
407 const VDataTool *tool = VAbstractPattern::getTool(history->at(i).getId());
408 if (tool != nullptr)
409 {
410 evalData = tool->getData();
411 break;
412 }
413 }
414 catch (const VExceptionBadId &)
415 {
416 continue;
417 }
418 }
419
420 evalData.RemoveVariable(currentLength);
421 evalData.RemoveVariable(currentSeamAllowance);
422 return evalData;
423}

425//---------------------------------------------------------------------------------------------------------------------
426void FinalMeasurementsDialog::fillTable()
427{
428 QTableWidget *table = ui->measurements_TableWidget;
429 const bool sortingEnabled = table->isSortingEnabled();
430
431 table->blockSignals(true);
432 table->setSortingEnabled(false);
433 table->clearContents();
434 table->setRowCount(m_measurements.size());
435
436 for (int i = 0; i < m_measurements.size(); ++i)
437 {
438 for (int column = ColumnName; column <= ColumnFormula; ++column)
439 {
440 QTableWidgetItem *item = new QTableWidgetItem();
441 item->setFlags(item->flags() ^ Qt::ItemIsEditable);
442 item->setData(Qt::UserRole, i);
443 if (column == ColumnValue)
444 {
445 item->setTextAlignment(Qt::AlignHCenter | Qt::AlignVCenter);
446 }
447 table->setItem(i, column, item);
448 }
449 updateRow(i);
450 }
451
452 table->setSortingEnabled(sortingEnabled);
453 table->blockSignals(false);
454
455 table->resizeColumnToContents(ColumnName);
456 table->resizeColumnToContents(ColumnValue);
457 table->resizeRowsToContents();
458
459 filterMeasurements(ui->filter_LineEdit->text());
460}

506//---------------------------------------------------------------------------------------------------------------------
507int FinalMeasurementsDialog::currentIndex() const
508{
509 const QTableWidget *table = ui->measurements_TableWidget;
510 const int row = table->currentRow();
511 if (row < 0 || table->selectedItems().isEmpty())
512 {
513 return -1;
514 }
515
516 const QTableWidgetItem *item = table->item(row, ColumnName);
517 if (item == nullptr)
518 {
519 return -1;
520 }
521
522 const int index = item->data(Qt::UserRole).toInt();
523 return (index >= 0 && index < m_measurements.size()) ? index : -1;
524}

569bool FinalMeasurementsDialog::isNameValid(const QString &name, int index) const
570{
571 if (name.isEmpty() || !QRegularExpression(QStringLiteral("^") + NameRegExp() + QStringLiteral("$")).match(name)
572 .hasMatch())
573 {
574 return false;
575 }
576
577 if (m_evalData.DataVariables()->contains(name))
578 {
579 return false;
580 }
581
582 for (int i = 0; i < m_measurements.size(); ++i)
583 {
584 if (i != index && m_measurements.at(i).name == name)
585 {
586 return false;
587 }
588 }
589 return true;
590}
