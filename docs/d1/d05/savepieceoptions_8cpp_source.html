85void SavePieceOptions::undo()
86{
87 qCDebug(vUndo, "Undo.");
88
89 QDomElement domElement = doc->elementById(nodeId, VAbstractPattern::TagPiece);
90 if (domElement.isElement())
91 {
92 PatternPieceTool::addAttributes(doc, domElement, nodeId, m_oldPiece);
93 doc->RemoveAllChildren(domElement);//Very important to clear before rewrite
94 PatternPieceTool::addPieceLabel(doc, domElement, m_oldPiece);
95 PatternPieceTool::addPatternLabel(doc, domElement, m_oldPiece);
96 PatternPieceTool::addGrainline(doc, domElement, m_oldPiece);
97 PatternPieceTool::addNodes(doc, domElement, m_oldPiece);
98 PatternPieceTool::addCSARecords(doc, domElement, m_oldPiece.getCustomSARecords());
99 PatternPieceTool::addInternalPaths(doc, domElement, m_oldPiece.getInternalPaths());
100 PatternPieceTool::addAnchors(doc, domElement, m_oldPiece.getAnchors());
101
102 IncrementReferences(m_oldPiece.MissingNodes(m_newPiece));
103 IncrementReferences(m_oldPiece.MissingCSAPath(m_newPiece));
104 IncrementReferences(m_oldPiece.MissingInternalPaths(m_newPiece));
105 IncrementReferences(m_oldPiece.missingAnchors(m_newPiece));
106
107 DecrementReferences(m_newPiece.MissingNodes(m_oldPiece));
108 DecrementReferences(m_newPiece.MissingCSAPath(m_oldPiece));
109 DecrementReferences(m_newPiece.MissingInternalPaths(m_oldPiece));
110 DecrementReferences(m_newPiece.missingAnchors(m_oldPiece));
111
112 emit NeedLiteParsing(Document::LiteParse);
113
114 if (auto *tool = qobject_cast<PatternPieceTool *>(VAbstractPattern::getTool(nodeId)))
115 {
116 tool->updatePiece(m_oldPiece);
117 }
118 }
119 else
120 {
121 qCWarning(vUndo, "Can't find piece with id = %u.", nodeId);
122 return;
123 }
124}

169bool SavePieceOptions::mergeWith(const QUndoCommand *command)
170{
171 const SavePieceOptions *saveCommand = static_cast<const SavePieceOptions *>(command);
172 SCASSERT(saveCommand != nullptr);
173 const quint32 id = saveCommand->pieceId();
174
175 if (id != nodeId)
176 {
177 return false;
178 }
179
180 m_newPiece = saveCommand->getNewPiece();
181 return true;
182}
