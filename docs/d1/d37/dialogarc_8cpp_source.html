439//---------------------------------------------------------------------------------------------------------------------
440void DialogArc::pointNameChanged()
441{
442 QColor color = okColor;
443
444 flagError = true;
445 color = okColor;
446
447 if (getCurrentObjectId(ui->centerPoint_ComboBox) == m_arc.GetCenter().id())
448 {
449 m_newDuplicate = -1;
450 ui->name_LineEdit->setText(qApp->translateVariables()->VarToUser(m_arc.name()));
451 }
452 else
453 {
454 VArc arc(*data->GeometricObject<VPointF>(getCurrentObjectId(ui->centerPoint_ComboBox)),
455 getRadius().toDouble(),
456 getF1().toDouble(),
457 getF2().toDouble());
458
459 if (!data->IsUnique(arc.name()))
460 {
461 m_newDuplicate = static_cast<qint32>(DNumber(arc.name()));
462 arc.SetDuplicate(static_cast<quint32>(m_newDuplicate));
463 }
464 if (m_arc.id() == NULL_ID)
465 {
466 ui->name_LineEdit->setText(qApp->translateVariables()->VarToUser(arc.name() + "_" + QString().setNum(m_Id + 1)));
467 }
468 else
469 {
470 ui->name_LineEdit->setText(qApp->translateVariables()->VarToUser(arc.name() + "_" + QString().setNum(m_arc.id())));
471 }
472 }
473
474 ChangeColor(ui->name_Label, color);
475 ChangeColor(ui->centerPoint_Label, color);
476 CheckState();
477}

490//---------------------------------------------------------------------------------------------------------------------
491/// @brief evalRadius calculate value of radius
492//---------------------------------------------------------------------------------------------------------------------
493void DialogArc::evalRadius()
494{
495 labelEditFormula = ui->labelEditRadius;
496 const QString postfix = UnitsToStr(qApp->patternUnit(), true);
497 const qreal radius = Eval(ui->plainTextEditFormula->toPlainText(), m_flagRadius, ui->labelResultRadius, postfix);
498
499 if (radius < 0)
500 {
501 m_flagRadius = false;
502 ChangeColor(labelEditFormula, Qt::red);
503 ui->labelResultRadius->setText(tr("Error"));
504 ui->labelResultRadius->setToolTip(tr("Radius can't be negative"));
505
506 DialogArc::CheckState();
507 }
508}

510//---------------------------------------------------------------------------------------------------------------------
511/// @brief evalAngles1 calculate value of angle
512//---------------------------------------------------------------------------------------------------------------------
513void DialogArc::evalAngles()
514{
515 labelEditFormula = ui->labelEditF1;
516 m_angle1 = Eval(ui->plainTextEditF1->toPlainText(), m_flagAngle1, ui->labelResultF1, degreeSymbol, false);
517
518 labelEditFormula = ui->labelEditF2;
519 m_angle2 = Eval(ui->plainTextEditF2->toPlainText(), m_flagAngle2, ui->labelResultF2, degreeSymbol, false);
520
521 checkAngles();
522}

524//---------------------------------------------------------------------------------------------------------------------
525void DialogArc::checkAngles()
526{
527 if (static_cast<int>(m_angle1) == INT_MIN || static_cast<int>(m_angle2) == INT_MIN)
528 {
529 return;
530 }
531
532 if (VFuzzyComparePossibleNulls(m_angle1, m_angle2))
533 {
534 m_flagAngle1 = false;
535 ChangeColor(ui->labelEditF1, Qt::red);
536 ui->labelResultF1->setText(tr("Error"));
537 ui->labelResultF1->setToolTip(tr("Angles equal"));
538
539 m_flagAngle2 = false;
540 ChangeColor(ui->labelEditF2, Qt::red);
541 ui->labelResultF2->setText(tr("Error"));
542 ui->labelResultF2->setToolTip(tr("Angles equal"));
543 }
544
545 DialogArc::CheckState();
546}
