330 bool VLayoutPaper::arrangePiece(const VLayoutPiece &piece, std::atomic_bool &stop)
331 {
332  // First need set size of paper
333  if (d->globalContour.GetHeight() <= 0 || d->globalContour.GetWidth() <= 0)
334  {
335  return false;
336  }
337 
338  if (piece.LayoutEdgesCount() < 3 || piece.pieceEdgesCount() < 3)
339  {
340  return false;//Not enough edges
341  }
342 
343  if (piece.IsForbidFlipping() && !d->globalRotate)
344  { // Compensate forbidden flipping by rotating. 180 degree will be enough.
345  d->localRotate = true;
346  d->localRotationIncrease = 180;
347  }
348  else
349  { // Return to global values if was changed
350  d->localRotate = d->globalRotate;
351  d->localRotationIncrease = d->globalRotationIncrease;
352  }
353 
354  return addToSheet(piece, stop);
355 }

369 //---------------------------------------------------------------------------------------------------------------------
370 /// @brief addToSheet add piece to layout page.
371 ///
372 /// This method adds a piece to a layout page.
373 ///
374 /// @param piece piece being added to page.
375 /// @param stop shared bool between threads.
376 /// @returns bool true if valid result, false if invalid result.
377 //---------------------------------------------------------------------------------------------------------------------
378 bool VLayoutPaper::addToSheet(const VLayoutPiece &piece, std::atomic_bool &stop)
379 {
380  VBestSquare bestResult(d->globalContour.GetSize(), d->saveLength);
381  QThreadPool *thread_pool = QThreadPool::globalInstance();
382  thread_pool->setExpiryTimeout(1000);
383  QVector<VPosition *> threads;
384 
385  int pieceEdgesCount = 0;
386 
387  if (d->globalContour.GetContour().isEmpty())
388  {
389  pieceEdgesCount = piece.pieceEdgesCount();
390  }
391  else
392  {
393  pieceEdgesCount = piece.LayoutEdgesCount();
394  }
395 
396  for (int j=1; j <= d->globalContour.GlobalEdgesCount(); ++j)
397  {
398  for (int i=1; i<= pieceEdgesCount; ++i)
399  {
400  VPosition *thread = new VPosition(d->globalContour, j, piece, i, &stop, d->localRotate,
401  d->localRotationIncrease,
402  d->saveLength);
403  thread->setAutoDelete(false);
404  threads.append(thread);
405  thread_pool->start(thread);
406  }
407  }
408 
409  // Wait for done
410  do
411  {
412  QCoreApplication::processEvents();
413  QThread::msleep(250);
414  }
415  while(thread_pool->activeThreadCount() > 0 && !stop.load());
416 
417  if (stop.load())
418  {
419  qDeleteAll(threads.begin(), threads.end());
420  threads.clear();
421  return false;
422  }
423 
424  for (int i=0; i < threads.size(); ++i)
425  {
426  bestResult.NewResult(threads.at(i)->getBestResult());
427  }
428 
429  qDeleteAll(threads.begin(), threads.end());
430  threads.clear();
431 
432  return saveResult(bestResult, piece);
433 }

436 bool VLayoutPaper::saveResult(const VBestSquare &bestResult, const VLayoutPiece &piece)
437 {
438  if (bestResult.ValidResult())
439  {
440  VLayoutPiece workDetail = piece;
441  workDetail.setTransform(bestResult.Transform());// Don't forget set transform
442  workDetail.SetMirror(bestResult.isMirror());
443  const QVector<QPointF> newGContour = d->globalContour.UniteWithContour(workDetail, bestResult.GContourEdge(),
444  bestResult.pieceEdge(),
445  bestResult.Type());
446  if (newGContour.isEmpty())
447  {
448  return false;
449  }
450  d->pieces.append(workDetail);
451  d->globalContour.SetContour(newGContour);
452  }
453 
454  return bestResult.ValidResult(); // Do we have the best result?
455 }

459 ///
460 /// This method returns a paper (page) as a QGraphicsRectItem.
461 ///
462 /// @param autoCrop If true autocrop the page, if false otherwise.

466 QGraphicsRectItem *VLayoutPaper::getPaperItem(bool autoCrop, bool textAsPaths) const
467 {
468  QGraphicsRectItem *paper;
469  if (autoCrop)
470  {
471  QScopedPointer<QGraphicsScene> scene(new QGraphicsScene());
472  QList<QGraphicsItem *> list = getPieceItems(textAsPaths);
473  for (int i=0; i < list.size(); ++i)
474  {
475  scene->addItem(list.at(i));
476  }
477  const int height = scene->itemsBoundingRect().toRect().height();
478  if (d->globalContour.GetHeight() > height) //-V807
479  {
480  paper = new QGraphicsRectItem(QRectF(0, 0, d->globalContour.GetWidth(), height));
481  }
482  else
483  {
484  paper = new QGraphicsRectItem(QRectF(0, 0, d->globalContour.GetWidth(), d->globalContour.GetHeight()));
485  }
486  }
487  else
488  {
489  paper = new QGraphicsRectItem(QRectF(0, 0, d->globalContour.GetWidth(), d->globalContour.GetHeight()));
490  }
491  paper->setPen(QPen(Qt::black, 1));
492  paper->setBrush(QBrush(Qt::white));
493  return paper;
