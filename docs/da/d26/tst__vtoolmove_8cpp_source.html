53void TST_VToolMove::explicitOriginIsNotTranslatedByTheMove()
54{
55 const Unit unit = Unit::Cm;
56 QScopedPointer<VContainer> data(new VContainer(nullptr, &unit));
57
58 data->UpdateGObject(originPointId, new VPointF(rotationPointA5.x(), rotationPointA5.y(), "A5", 0, 0));
59 data->UpdateGObject(sourcePointId, new VPointF(-6.432290, 8.464600, "A1", 0, 0));
60
61 const QPointF origin = VToolMove::findRotationOrigin(QVector<quint32>{sourcePointId}, data.data(), moveLength,

89void TST_VToolMove::moveWithRotationMatchesRotationAfterMove()
90{
91 const Unit unit = Unit::Cm;
92 QScopedPointer<VContainer> data(new VContainer(nullptr, &unit));
93
94 data->UpdateGObject(originPointId, new VPointF(rotationPointA5.x(), rotationPointA5.y(), "A5", 0, 0));
95
96 QVector<VPointF> sourcePoints;
97 sourcePoints.append(VPointF(-6.432290, 8.464600, "A1", 0, 0));
98 sourcePoints.append(VPointF(3.567710, 8.464600, "A2", 0, 0));
99 sourcePoints.append(VPointF(3.567710, 18.464600, "A3", 0, 0));
100 sourcePoints.append(VPointF(13.567710, 18.464600, "A4", 0, 0));
101
102 QVector<quint32> objectIds;
103 for (int i = 0; i < sourcePoints.size(); ++i)
104 {
105 const quint32 id = sourcePointId + static_cast<quint32>(i);
106 data->UpdateGObject(id, new VPointF(sourcePoints.at(i)));
107 objectIds.append(id);
108 }
109
110 const QPointF origin = VToolMove::findRotationOrigin(objectIds, data.data(), moveLength, moveAngle,
111 originPointId);
112
113 for (auto point : sourcePoints)
114 {
115 // What the Move tool produces.
116 const QPointF moveAndRotate =
117 static_cast<QPointF>(point.Move(moveLength, moveAngle).Rotate(origin, rotationAngle));
118
119 // What the Rotation tool applied after a plain move produces - the equivalence the reporter
120 // demonstrated as correct.
121 const QPointF moveThenRotate =
122 static_cast<QPointF>(point.Move(moveLength, moveAngle).Rotate(rotationPointA5, rotationAngle));
123
124 QVERIFY2(VFuzzyComparePoints(moveAndRotate, moveThenRotate),
125 qUtf8Printable(QStringLiteral("Moving with a rotation must equal moving and then rotating about "
126 "the same point. Point %1: got (%2, %3), expected (%4, %5).")
127 .arg(point.name()).arg(moveAndRotate.x()).arg(moveAndRotate.y())
128 .arg(moveThenRotate.x()).arg(moveThenRotate.y())));
129 }
130}

133void TST_VToolMove::implicitOriginStillFollowsTheMove()
134{
135 const Unit unit = Unit::Cm;
136 QScopedPointer<VContainer> data(new VContainer(nullptr, &unit));
137
138 // Three points, deliberately asymmetric: their bounding-box centre (10, 10) differs from both their
139 // centroid (10, 6.667) and any pairwise midpoint, so this test actually pins "bounding-box centre"
140 // rather than passing under any of those other, superficially similar rules too.
141 data->UpdateGObject(sourcePointId, new VPointF(0, 0, "A1", 0, 0));
142 data->UpdateGObject(sourcePointId + 1, new VPointF(10, 20, "A2", 0, 0));
143 data->UpdateGObject(sourcePointId + 2, new VPointF(20, 0, "A3", 0, 0));
144
145 const QVector<quint32> objectIds{sourcePointId, sourcePointId + 1, sourcePointId + 2};
146 const QPointF origin = VToolMove::findRotationOrigin(objectIds, data.data(), moveLength, moveAngle, NULL_ID);
147
148 // Without an explicit rotation point the origin is the bounding box centre translated by the move
149 // vector. That behaviour is correct and must not change.
150 const QPointF expected = VPointF::MovePF(QPointF(10, 10), moveLength, moveAngle);
151
152 QVERIFY2(VFuzzyComparePoints(origin, expected),
153 qUtf8Printable(QStringLiteral("Without an explicit rotation point the origin must keep following the "
154 "move. Expected (%1, %2), got (%3, %4).")
155 .arg(expected.x()).arg(expected.y()).arg(origin.x()).arg(origin.y())));
156}
