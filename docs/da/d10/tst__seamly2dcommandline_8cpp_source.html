333void TST_Seamly2DCommandLine::automationDataDirectory()
334{
335 QFETCH(bool, seamlyMe);
336 QFETCH(QString, scenario);
337 QTemporaryDir directory;
338 QVERIFY(directory.isValid());
339 const QString dataDir = directory.filePath("state");
340 QString requested = dataDir;
341 QString protectedPath;
342 QFileDevice::Permissions originalPermissions;
343 const auto restorePermissions = qScopeGuard([&]()
344 {
345 if (!protectedPath.isEmpty())
346 {
347 QFile::setPermissions(protectedPath, originalPermissions);
348 }
349 });
350 int expected = V_EX_CANTCREAT;
351 if (scenario.startsWith("valid"))
352 {
353 expected = V_EX_OK;
354 // Exercise normal startup, settings sync, folders and logging; --version exits too early.
355 if (scenario == "valid")
356 {
357 QVERIFY(QDir().mkpath(dataDir + "/settings/Seamly2DTeam"));
358 QSettings settings(dataDir + "/settings/Seamly2DTeam/qt6_" +
359 (seamlyMe ? "seamlyme.ini" : "seamly2d.ini"), QSettings::IniFormat);
360 settings.setValue("configuration/locale", "en_US");
361 settings.setValue("automation/sentinel", "preserved");
362 settings.sync();
363 QCOMPARE(settings.status(), QSettings::NoError);
364 }
365 }
366 else if (scenario == "relative")
367 {
368 requested = "relative-state";
369 expected = V_EX_CONFIG;
370 }
371 else if (scenario.startsWith("file-"))
372 {
373 QString target = dataDir;
374 if (scenario != "file-root")
375 {
376 QVERIFY(QDir().mkpath(dataDir));
377 target += scenario == "file-settings" ? "/settings" : "/logs";
378 }
379 QFile file(target);
380 QVERIFY(file.open(QIODevice::WriteOnly));
381 file.close();
382 }
383 else
384 {
385 protectedPath = dataDir + (scenario == "readonly-settings" ? "/settings" : "/logs/seamly2d");
386 // mkpath alone would pass because all directories already exist.
387 QVERIFY(QDir().mkpath(dataDir + "/settings/Seamly2DTeam"));
388 QVERIFY(QDir().mkpath(dataDir + "/logs/seamly2d"));
389 originalPermissions = QFileInfo(protectedPath).permissions();
390 QVERIFY(QFile::setPermissions(protectedPath, QFileDevice::ReadOwner | QFileDevice::ExeOwner));
391 QTemporaryFile probe(protectedPath + "/test-XXXXXX");
392 if (probe.open())
393 {
394 QSKIP("This account/filesystem can still write despite read-only bits (e.g. Windows ACLs/root).");
395 }
396 }
397
398 QProcess process;
399 process.setWorkingDirectory(directory.path());
400 QProcessEnvironment environment = QProcessEnvironment::systemEnvironment();
401 environment.insert("SEAMLY_DATA_DIR", requested);
402 environment.insert("QT_QPA_PLATFORM", "offscreen");
403 environment.insert("LANG", "C");
404 environment.insert("LC_ALL", "C");
405 process.setProcessEnvironment(environment);
406 const QString fixture = QCoreApplication::applicationDirPath() + QDir::separator() +
407 (seamlyMe ? QString("tst_seamlyme/empty.smis") : tmpTestCollectionFolder + "/bra.sm2d");
408 process.start(seamlyMe ? SeamlyMePath() : Seamly2DPath(), QStringList() << "--test" << fixture);
409 QVERIFY2(process.waitForFinished(60000), qPrintable(process.errorString()));
410 QCOMPARE(process.exitStatus(), QProcess::NormalExit);
411 const QByteArray errors = process.readAllStandardError();
412 QVERIFY2(process.exitCode() == expected, errors.constData());
413 if (expected != V_EX_OK)
414 {
415 QVERIFY(errors.contains("SEAMLY_DATA_DIR"));
416 QVERIFY(!QDir(directory.filePath("relative-state")).exists());
417 return;
418 }
419
420 const QStringList folders = {"templates", "measurements/individual", "measurements/multisize", "label templates"};
421 for (const QString &folder : folders)
422 {
423 QVERIFY2(QDir(dataDir + "/" + folder).exists(), qPrintable(folder));
424 }
425 if (scenario == "valid")
426 {
427 QSettings saved(dataDir + "/settings/Seamly2DTeam/qt6_" +
428 (seamlyMe ? "seamlyme.ini" : "seamly2d.ini"), QSettings::IniFormat);
429 QCOMPARE(saved.value("automation/sentinel").toString(), QString("preserved"));
430 }
431 if (!seamlyMe)
432 {
433 QVERIFY(QDir(dataDir + "/patterns").exists());
434 QVERIFY(QDir(dataDir + "/layouts").exists());
435 QVERIFY(QDir(dataDir + "/backups").exists());
436 const QDir logs(dataDir + "/logs/seamly2d");
437 const QStringList files = logs.entryList(QStringList() << "seamly2d-pid*.log", QDir::Files);
438 QCOMPARE(files.size(), 1);
439 QFile log(logs.filePath(files.first()));
440 QVERIFY(log.open(QIODevice::ReadOnly));
441 QVERIFY(log.readAll().contains("Command-line arguments:"));
442 }
443}
