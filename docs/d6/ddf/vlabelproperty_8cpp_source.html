83QWidget *VPE::VLabelProperty::createEditor(QWidget *parent, const QStyleOptionViewItem &options,
84 const QAbstractItemDelegate *delegate)

87 Q_UNUSED(delegate)
88

92 labelTextEditor->setSizePolicy(QSizePolicy::Expanding, QSizePolicy::Fixed);
