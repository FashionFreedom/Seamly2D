82void SavePiecePathOptions::undo()
83{
84 qCDebug(vUndo, "Undo.");
85
86 QDomElement domElement = doc->elementById(nodeId, VAbstractPattern::TagPath);
87 if (domElement.isElement())
88 {
89 InternalPathTool::addAttributes(doc, domElement, nodeId, m_oldPath);
90 doc->RemoveAllChildren(domElement);//Very important to clear before rewrite
91 InternalPathTool::addNodes(doc, domElement, m_oldPath);
92
93 IncrementReferences(m_oldPath.MissingNodes(m_newPath));
94
95 SCASSERT(m_data);
96 m_data->UpdatePiecePath(nodeId, m_oldPath);
97
98 if (m_pieceId != NULL_ID)
99 {
100 if (PatternPieceTool *piece = qobject_cast<PatternPieceTool *>(VAbstractPattern::getTool(m_pieceId)))
101 {
102 piece->RefreshGeometry();
103 }
104 }
105 }
106 else
107 {
108 qCWarning(vUndo, "Can't find path with id = %u.", nodeId);
109 }
110}

144bool SavePiecePathOptions::mergeWith(const QUndoCommand *command)
145{
146 const SavePiecePathOptions *saveCommand = static_cast<const SavePiecePathOptions *>(command);
147 SCASSERT(saveCommand != nullptr);
148 const quint32 id = saveCommand->pathId();
149
150 if (id != nodeId)
151 {
152 return false;
153 }
154
155 m_newPath = saveCommand->newPath();
156 return true;
157}
