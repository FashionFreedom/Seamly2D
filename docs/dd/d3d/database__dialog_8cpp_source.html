263/// @brief eventFilter Monitors keyboard navigation and widget container resizing events.
264///
265/// Intercepts directional up/down arrow key presses within the measurement selection tree widget to
266/// synchronize and trigger preview updates immediately before selection transitions finalize. It also
267/// intercepts geometric changes to the graphic frame container to force real-time vector image scaling.
268///
269/// @param object The object currently being monitored by the filter layout.
270/// @param event The layout or input event structure being dispatched.
271/// @return bool True if the event was handled and consumed; otherwise returns the base class result.
272//---------------------------------------------------------------------------------------------------------------------
273bool MeasurementDatabaseDialog::eventFilter(QObject *object, QEvent *event)
274{
275 if (object == ui->treeWidget && event->type() == QEvent::KeyPress)
276 {
277 QKeyEvent *keyEvent = static_cast<QKeyEvent *>(event);
278 switch(keyEvent->key())
279 {
280 case Qt::Key_Up:
281 {
282 const QModelIndex model = ui->treeWidget->indexAbove(ui->treeWidget->currentIndex());
283 QTreeWidgetItem *item = ui->treeWidget->itemAbove(ui->treeWidget->currentItem());
284 showDescription(item, model.column());
285 break;
286 }
287 case Qt::Key_Down:
288 {
289 const QModelIndex model = ui->treeWidget->indexBelow(ui->treeWidget->currentIndex());
290 QTreeWidgetItem *item = ui->treeWidget->itemBelow(ui->treeWidget->currentItem());
291 showDescription(item, model.column());
292 break;
293 }
294 default:
295 break;
296 }
297 }
298 else if (object == ui->diagram_Frame && event->type() == QEvent::Resize)
299 {
300 renderScaledDiagram();
301 }
302 return QDialog::eventFilter(object, event);
303}

306/// @brief updateChecks Synchronizes multi-state checkbox flags recursively through the database tree.
307///
308/// Evaluates user modifications to tree item selection states. When a parent category checkbox state changes,
309/// it automatically ripples those assignments down to uncommitted child leaves. For individual leaf item modifications,
310/// it commits or rolls back entries from tracking data arrays and recursively updates parent branches to
311/// Checked, Unchecked, or PartiallyChecked layout flags.
312///
313/// @param item The tree widget item containing a selection state that needs tracking or propagation.
314/// @param column The zero-based column position, or a fallback flag (-1) indicating internal recursive execution.
315//---------------------------------------------------------------------------------------------------------------------
316void MeasurementDatabaseDialog::updateChecks(QTreeWidgetItem *item, int column)
317{
318 bool diff = false;
319 if (column != 0 && column != -1)
320 {
321 return;
322 }
323
324 if (item->childCount() != 0 && item->checkState(0) != Qt::PartiallyChecked && column != -1)
325 {
326 bool flag = false; // Check if we could change atleast one children
327 Qt::CheckState state = item->checkState(0);
328 for (int i = 0; i < item->childCount(); ++i)
329 {
330 if (!m_measurements.contains(item->child(i)->data(0, Qt::UserRole).toString()))
331 {
332 item->child(i)->setCheckState(0, state);
333 flag = true;
334 }
335
336 if (flag == false) // All child in the list
337 {
338 item->setCheckState(0, Qt::Checked);
339 }
340 }
341 }
342 else if (item->childCount() == 0 || column == -1)
343 {
344
345 QString itemName = item->data(0, Qt::UserRole).toString();
346 Qt::CheckState state = item->checkState(0);
347
348 if (state == Qt::CheckState::Unchecked)
349 {
350 m_newMeasurements.removeOne(itemName);
351 }
352 else if (state == Qt::CheckState::Checked &&
353 !m_measurements.contains(itemName) &&
354 !m_newMeasurements.contains(itemName) &&
355 !itemName.isEmpty())
356 {
357 m_newMeasurements.append(itemName);
358 }
359
360 QTreeWidgetItem *parent = item->parent();
361 if (parent == nullptr)
362 {
363 return;
364 }
365 for (int j = 0; j < parent->childCount(); ++j)
366 {
367 if (j != parent->indexOfChild(item)
368 && item->checkState(0) != parent->child(j)->checkState(0))
369 {
370 diff = true;
371 }
372 }
373 if (diff)
374 {
375 parent->setCheckState(0, Qt::PartiallyChecked);
376 }
377 else
378 {
379 parent->setCheckState(0, item->checkState(0));
380 }
381
382 updateChecks(parent, -1);
383 }
384}

387/// @brief showDescription Processes tree item selection changes to load and display the matching measurement diagram.
388///
389/// Validates the active tree item and column selection index to ensure only valid, childless leaf nodes
390/// are processed. For qualified selections, it extracts the underlying measurement data string, maps it
391/// to its corresponding vector graphic index, and caches the layout path state. It then triggers an initial
392/// high-resolution rendering pass or falls back to a warning view if the index is unknown.
393///
394/// @param item The tree widget item row that was selected or clicked.
395/// @param column The zero-based column position where the selection event occurred.
396//---------------------------------------------------------------------------------------------------------------------
397void MeasurementDatabaseDialog::showDescription(QTreeWidgetItem *item, int column)
398{
399 if ((column != 0 && column != -1) || item == nullptr || (item->childCount() != 0 ))
400 {
401 ui->diagram_Label->clear();
402 return;
403 }
404
405 const QString name = item->data(0, Qt::UserRole).toString();
406 const VTranslateVars *trv = qApp->translateVariables();
407 const QString number = trv->MNumber(name);
408 const QString description = trv->Description(name);
409
410 // Clear variables so resizes don't draw a stale image
411 m_currentSvgPath.clear();
412 m_currentNumber.clear();
413 m_currentName.clear();
414 m_currentDescription.clear();
415
416 // Save the current states for the resize event to look at later
417 m_currentSvgPath = QString("://diagrams/%1.svg").arg(MapDiagrams(trv, number));
418 m_currentNumber = number;
419 m_currentName = trv->guiText(name);
420 m_currentDescription = description;
421
422 // Execute the initial draw pass using the live frame size
423 renderScaledDiagram();
424}

576//---------------------------------------------------------------------------------------------------------------------
577/// @brief addMeasurement Compiles database tokens to allocate, format, and filter a child tree node.
578///
579/// Asserts that a valid parent node exists, then appends a new child element directly under it. If operating
580/// under selection mode rules, it references active array indexes to determine if the node is hard-locked, pre-checked,
581/// or unchecked—updating its flags, background highlight masks, and recursively balancing parent checkboxes. It concludes
582/// by translating internal data definitions into formatted user-facing measurement labels.
583///
584/// @param group A pointer to the direct parent category node where this item will reside.
585/// @param name The primary database dictionary key identifier string for this specific metric.
586/// @param list A verification dataset array used to dictate baseline checklist activations.
587//---------------------------------------------------------------------------------------------------------------------
588void MeasurementDatabaseDialog::addMeasurement(QTreeWidgetItem *group, const QString &name, const QStringList &list)
589{
590 SCASSERT(group != nullptr)
591
592 QTreeWidgetItem *item = new QTreeWidgetItem(group);
593
594 if (m_selectMode)
595 {
596 if (m_measurements.contains(name))
597 {
598 item->setFlags(Qt::ItemIsSelectable | Qt::ItemIsEnabled);
599 item->setCheckState(0, Qt::Checked);
600 item->setBackground(0, QBrush(QColor(231, 231, 0, 127)));
601 }
602 else if (list.contains(name))
603 {
604 item->setCheckState(0, Qt::Checked);
605 }
606 else
607 {
608 item->setCheckState(0, Qt::Unchecked);
609 }
610
611 updateChecks(item, 0);
612 }
613
614 const QString text = qApp->translateVariables()->MNumber(name) + ". " + qApp->translateVariables()->MToUser(name);
615 item->setText(0, text);
616 item->setToolTip(0, text);
617 item->setData(0, Qt::UserRole, name);
618}

643/// @brief filterGroups Dispatches text filtering queries to evaluate node visibility across all tree categories.

651void MeasurementDatabaseDialog::filterGroups(const QString searchString)

678/// Asserts that a valid target folder pointer exists, then loops through its direct leaf children to verify if their

686{

691 {
692 QTreeWidgetItem *item = group->child(i);
693 const bool isHidden = !item->text(0).contains(searchString, Qt::CaseInsensitive);
694
695 item->setHidden(isHidden);
696 if (!isHidden)
697 {
698 isMatch = true;
699 }
700 }
701
702 group->setHidden(!group->text(0).contains(searchString, Qt::CaseInsensitive) && !isMatch);
703}

842/// @brief renderScaledDiagram Generates a pixel-perfect, crisp vector diagram to fit the current frame size.
843///
844/// Loads the active SVG file path from memory and dynamically reads the real-time dimensions of the
845/// main window's frame container. It compares horizontal and vertical scaling factors to determine the
846/// maximum aspect-ratio fit, preventing layout bleeding. The mathematical vector paths are then rendered
847/// onto a sharp, transparent canvas layout, updating the display widget alongside its text caption label.
848//---------------------------------------------------------------------------------------------------------------------
849void MeasurementDatabaseDialog::renderScaledDiagram()
850{
851 if (m_currentSvgPath.isEmpty()) return;
852
853 QSvgRenderer renderer(m_currentSvgPath);
854 QSize nativeSize = renderer.defaultSize();
855
856 if (nativeSize.isValid() && nativeSize.width() > 0 && nativeSize.height() > 0)
857 {
858 // Read from the live frame container dimensions. Falling back to safe defaults if frame hasn't painted yet.
859 double maxW = (ui->diagram_Frame->width() > 0) ? (ui->diagram_Frame->width() - 10.0) : 290.0;
860 double maxH = (ui->diagram_Frame->height() > 0) ? (ui->diagram_Frame->height() - 120.0) : 500.0;
861
862 double ratioW = maxW / nativeSize.width();
863 double ratioH = maxH / nativeSize.height();
864 double scaleFactor = (ratioW < ratioH) ? ratioW : ratioH;
865
866 int targetWidth = static_cast<int>(nativeSize.width() * scaleFactor);
867 int targetHeight = static_cast<int>(nativeSize.height() * scaleFactor);
868
869 QPixmap pixmap(targetWidth, targetHeight);
870 pixmap.fill(Qt::transparent);
871
872 QPainter painter(&pixmap);
873 renderer.render(&painter);
874 painter.end();
875
876 ui->diagram_Label->setPixmap(pixmap);
877 }
878
879 ui->caption_Label->setText(QString("<b>%1</b>. <i>%2</i>").arg(m_currentNumber, m_currentName));
880 ui->description_Label->setText(m_currentDescription);
881}
