64/// @brief The NumericItem class sorts by the number stored in Qt::UserRole instead of the text, so "100.5" sorts after
65/// "20.3". Cells without a number (-1) sort first.
66///
67/// @param text Text name of the QTableWidgetItem
68/// @param value Data of the QTableWidgetItem
69//---------------------------------------------------------------------------------------------------------------------
70class NumericItem : public QTableWidgetItem
71{
72public:
73 NumericItem(const QString &text, qreal value)
74 : QTableWidgetItem(text)
75 {
76 setData(Qt::UserRole, value);
77 setTextAlignment(Qt::AlignRight | Qt::AlignVCenter);
78 }
79
80 virtual bool operator<(const QTableWidgetItem &other) const override
81 {
82 return data(Qt::UserRole).toDouble() < other.data(Qt::UserRole).toDouble();
83 }
84};

87/// @brief numberLocale set the locale number options to omit the group separtors.
88//---------------------------------------------------------------------------------------------------------------------
89QLocale numberLocale()
90{
91 // No thousands separators: the values are copied and exported to spreadsheets, which may not read them back.
92 QLocale locale = qApp->Settings()->getOsSeparator() ? QLocale() : QLocale::c();
93 locale.setNumberOptions(locale.numberOptions() | QLocale::OmitGroupSeparator);
94 return locale;
95}

98/// @brief Creates a numeric table item displaying a length converted from pixels to the application's current unit.
99///
100/// @param pixels The length in pixels. Returns an empty item with a sort value of -1 if negative.
101/// @return A pointer to the newly created NumericItem.
102//---------------------------------------------------------------------------------------------------------------------
103QTableWidgetItem *lengthItem(qreal pixels)
104{
105 if (pixels < 0)
106 {
107 return new NumericItem(QString(), -1);
108 }
109 const qreal value = qApp->fromPixel(pixels);
110 return new NumericItem(numberLocale().toString(value, 'f', 2), value);
111}

136/// @brief fillTable fill table
137//---------------------------------------------------------------------------------------------------------------------
138void PieceInfoDialog::fillTable()
139{
140 const QString unit = UnitsToStr(qApp->patternUnit(), true);
141 const QHash<quint32, VPiece> *pieces = m_data->DataPieces();
142
143 QTableWidget *table = ui->pieces_TableWidget;
144 QString seam_line_str = tr("Seam Line");
145 QString cut_line_str = tr("Cut Line");
146 QString length_around_str = tr("Length around the piece ");
147
148 table->setSortingEnabled(false);
149 table->setColumnCount(ColumnCount);
150 table->setRowCount(pieces->size());
151 table->setHorizontalHeaderLabels({tr("Name"), tr("Letter"), tr("Quantity"),
152 tr("Seam Allowance (%1)").arg(unit),
153 seam_line_str + QString("(%1)").arg(unit),
154 cut_line_str + QString("(%1)").arg(unit)});
155 table->horizontalHeaderItem(ColumnSeamLine)->setToolTip(length_around_str + seam_line_str);
156 table->horizontalHeaderItem(ColumnCutLine)->setToolTip(length_around_str + cut_line_str);
157
158 int row = 0;
159 for (auto i = pieces->constBegin(); i != pieces->constEnd(); ++i, ++row)
160 {
161 const VPiece &piece = i.value();
162 const VPieceLabelData &label = piece.GetPatternPieceData();
163
164 // Name
165 table->setItem(row, ColumnName, new QTableWidgetItem(piece.GetName()));
166
167 // Pattern letter
168 QTableWidgetItem *letter_item = new QTableWidgetItem(label.GetLetter());
169 letter_item->setTextAlignment(Qt::AlignHCenter | Qt::AlignVCenter);
170 table->setItem(row, ColumnLetter, letter_item);
171
172 // Piece quantity
173 QTableWidgetItem *quantity = new NumericItem(QString::number(label.GetQuantity()), label.GetQuantity());
174 if (label.IsOnFold())
175 {
176 quantity->setToolTip(tr("Cut on fold"));
177 }
178 table->setItem(row, ColumnQuantity, quantity);
179
180 // Seam allowance
181 QTableWidgetItem *seam_allowance;
182 if (piece.hasSeamAllowanceBuiltIn())
183 {
184 seam_allowance = new NumericItem(tr("Built in"), -1);
185 }
186 else if (piece.hasSeamAllowance())
187 {
188 const qreal sa_width = piece.GetSAWidth();
189 seam_allowance = new NumericItem(numberLocale().toString(sa_width, 'f', 2), sa_width);
190 }
191 else
192 {
193 seam_allowance = new NumericItem(tr("None"), 0);
194 }
195 table->setItem(row, ColumnSeamAllowance, seam_allowance);
196
197 // seam line length
198 qreal seam_line = -1;
199
200 QTableWidgetItem *seam_line_item;
201 if (!piece.hasSeamAllowanceBuiltIn())
202 {
203 seam_line = piece.seamlineLength(m_data);
204 seam_line_item = lengthItem(seam_line);
205 }
206 else
207 {
208 // With the seam allowance built in, the piece's outline is the cutting line and its seamline isn't drawn.
209 seam_line_item = new NumericItem(QString("NA"), -1);
210 seam_line_item->setTextAlignment(Qt::AlignRight | Qt::AlignVCenter);
211 }
212 table->setItem(row, ColumnSeamLine, seam_line_item);
213
214 // Cut line length
215 qreal cut_line = -1;
216 cut_line = piece.cutPathLength(m_data);
217 table->setItem(row, ColumnCutLine, lengthItem(cut_line));
218 }
219
220 table->setSortingEnabled(true);
221 table->sortByColumn(ColumnName, Qt::AscendingOrder);
222 table->resizeColumnsToContents();
223 table->horizontalHeader()->setStretchLastSection(true);
224}

254/// @brief copyToClipboard copy dialog selection to clipboard as comma separated values.
255//---------------------------------------------------------------------------------------------------------------------
256void PieceInfoDialog::copyToClipboard()
257{
258 QItemSelectionModel *model = ui->pieces_TableWidget->selectionModel();
259 QModelIndexList selected_indexes = model->selectedIndexes();
260
261 QString clipboard_string;
262
263 for (int i = 0; i < selected_indexes.count(); ++i)
264 {
265 QModelIndex current = selected_indexes[i];
266 QString display_text = current.data(Qt::DisplayRole).toString();
267
268 // Check if another column exists beyond this one.
269 if (i + 1 < selected_indexes.count())
270 {
271 QModelIndex next = selected_indexes[i+1];
272
273 // If the column is on different row, the clipboard should take note.
274 if (next.row() != current.row())
275 {
276 display_text.append("\n");
277 }
278 else
279 {
280 // Otherwise append a comma separator.
281 display_text.append(" , ");
282 }
283 }
284 clipboard_string.append(display_text);
285 }
286
287 QClipboard *clipboard = QApplication::clipboard();
288 clipboard->setText(clipboard_string);
289}

292/// @brief exportToCsv saves the table as a CSV file, with the separator, encoding and header chosen as for exporting
293/// measurements.
294//---------------------------------------------------------------------------------------------------------------------
295void PieceInfoDialog::exportToCsv()
296{
297 const QString suffix = QStringLiteral("csv");
298 QString name = QFileInfo(qApp->getFilePath()).completeBaseName();
299 if (name.isEmpty())
300 {
301 name = tr("pieces");
302 }
303 const QString path = QDir::homePath() + QLatin1Char('/') + name + QLatin1Char('.') + suffix;
304
305 QString file_name = QFileDialog::getSaveFileName(this, tr("Export to CSV"), path,
306 tr("Comma-Separated Values") + QLatin1String(" (*.csv)"),
307 nullptr, qApp->Settings()->getUseNativeFileDialogs());
308 if (file_name.isEmpty())
309 {
310 return;
311 }
312 if (QFileInfo(file_name).suffix().isEmpty())
313 {
314 file_name += QLatin1Char('.') + suffix;
315 }
316
317 DialogExportToCSV dialog(this);
318 if (dialog.exec() != QDialog::Accepted)
319 {
320 return;
321 }
322
323 QxtCsvModel csv;
324 const QStringList header = headerTexts();
325 for (int column = 0; column < header.size(); ++column)
326 {
327 csv.insertColumn(column);
328 if (dialog.WithHeader())
329 {
330 csv.setHeaderText(column, header.at(column));
331 }
332 }
333 for (int row = 0; row < ui->pieces_TableWidget->rowCount(); ++row)
334 {
335 csv.insertRow(row);
336 const QStringList texts = rowTexts(row);
337 for (int column = 0; column < texts.size(); ++column)
338 {
339 csv.setText(row, column, texts.at(column));
340 }
341 }
342 csv.toCSV(file_name, dialog.WithHeader(), dialog.Separator(), dialog.SelectedEncoding());
343
344 qApp->Settings()->SetCSVSeparator(dialog.Separator());
345 qApp->Settings()->SetCSVCodec(dialog.SelectedEncoding());
346 qApp->Settings()->SetCSVWithHeader(dialog.WithHeader());
347}
