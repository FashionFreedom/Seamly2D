115inline void DOMAttrImpl::removeAttrFromIDNodeMap()
116{
117 if (fNode.isIdAttr()) {
118 ((DOMDocumentImpl *)fParent.fOwnerDocument)->getNodeIDMap()->remove(this);
119 fNode.isIdAttr(false);
120 }
121}
122
123inline void DOMAttrImpl::addAttrToIDNodeMap()
124{
125 if (fNode.isIdAttr())
126 return;
127
128 fNode.isIdAttr(true);
129
130 // REVIST For now, we don't worry about what happens if the new
131 // name conflicts as per setValue
132 DOMDocumentImpl *doc = (DOMDocumentImpl *)(fParent.fOwnerDocument);
133
134 if (doc->fNodeIDMap == 0)
135 doc->fNodeIDMap = new (doc) DOMNodeIDMap(500, doc);
136
137 doc->getNodeIDMap()->add(this);
138}
