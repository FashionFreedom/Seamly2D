40//
41// DStringPoolEntry - one of these structs is allocated for each
42// XMLCh String in the pool. Each slot in the
43// hash table array itself is a pointer to the head
44// of a singly-linked list of these structs.
45//
46// Although this struct is declared with a string length of one,
47// the factory method allocates enough storage to hold the full
48// string length.
49//
50struct DOMStringPoolEntry
51{
52 DOMStringPoolEntry *fNext;
53 XMLSize_t fLength;
54 XMLCh fString[1];
55};

155inline void DOMBuffer::
156append (const XMLCh* const chars)
157{
158 XMLSize_t count = XMLString::stringLen(chars);
159 if (fIndex + count >= fCapacity)
160 expandCapacity(count);
161
162 memcpy(&fBuffer[fIndex], chars, count * sizeof(XMLCh));
163 fIndex += count;
164
165 // Keep it null terminated
166 fBuffer[fIndex] = 0;
167}
